cqlib_qml.models

Quantum neural network models for hybrid quantum-classical learning.

This module provides quantum and hybrid quantum-classical neural network models that combine parameterized quantum circuits with classical layers. These models support automatic differentiation and are designed for various machine learning tasks.

Available Models: - Module: Base class for composing quantum and classical components. - QNN: Pure Quantum Neural Network using only quantum circuits. - HQNN: Hybrid Quantum-Classical Neural Network with a classical output layer.

The models support: - Automatic differentiation (backpropagation) - Parameter optimization - Batch processing of quantum circuits - Checkpoint saving and loading

Examples:

from cqlib_qml.models import QNN, HQNN from cqlib_qml.ansatz import HEAnsatz from cqlib_qml.encoder import FRQI

Pure quantum neural network

ansatz = HEAnsatz(n_qubits=4, d=2, layers=["RY", "CX"]) ansatz.set_measurement(readouts=[0]) qnn = QNN(ansatz=ansatz)

Hybrid quantum-classical neural network

ansatz = HEAnsatz(n_qubits=4, d=2, layers=["RY", "CX"]) hqnn = HQNN(ansatz=ansatz, out_dim=3)

 43class Module:
 44    """
 45    Container for composing quantum and classical components.
 46
 47    The Module class allows flexible composition of layers and ansätze
 48    into a single model. It handles forward and backward propagation
 49    through the composed components, parameter management, and
 50    checkpointing.
 51
 52    Supported composition patterns:
 53        1. linear + linear: Outputs of first linear layer are inputs to second
 54        2. ansatz + linear: Outputs of ansatz are inputs to linear layer
 55        3. linear + ansatz: Outputs of linear layer are parameters of ansatz
 56        4. ansatz + ansatz: Outputs of first ansatz are parameters of second
 57
 58    Args:
 59        *args (Union[Layer, Ansatz]): Variable number of layers and/or
 60            ansätze to compose.
 61
 62    Attributes:
 63        _nets (list): List of composed components.
 64
 65    Raises:
 66        TypeError: If any argument is not a Layer or Ansatz.
 67        ValueError: If an Ansatz doesn't have measurements.
 68
 69    Examples:
 70        >>> # Hybrid model
 71        >>> model = Module(ansatz, linear)
 72        >>>
 73        >>> # Classical model
 74        >>> model = Module(layer1, layer2)
 75        >>>
 76        >>> # Quantum model
 77        >>> model = Module(ansatz1, ansatz2)
 78    """

 87        Raises:
 88            TypeError: If any argument is not a Layer or Ansatz.
 89            ValueError: If an Ansatz doesn't have measurements.

 92            >>> model = Module(ansatz, linear)
 93            >>> model = Module(linear1, linear2)

 97    def forward(self, x=None):
 98        """
 99        Perform forward propagation through the module.
100
101        This method passes the input through each component in sequence,
102        where the output of one component becomes the input to the next.
103
104        Args:
105            x (np.ndarray, optional): Input data. Defaults to None.
106
107        Returns:
108            np.ndarray: Output of the last component in the module.
109
110        Examples:
111            >>> # With input data
112            >>> output = model.forward(X)
113            >>>
114            >>> # For ansatz-only models (no input needed)
115            >>> output = model.forward()
116        """
117        x_in = x
118        for net in self._nets:
119            x_in = net.forward(x_in)
120        return x_in
121
122    def backward(self, dLdout=None):
123        """
124        Perform backward propagation through the module.
125
126        This method propagates gradients backwards through the module,
127        from the last component to the first.
128
129        Args:
130            dLdout (Union[float, np.ndarray], optional): Gradients of the
131                loss with respect to the module's output. Required if the
132                last component is a classical layer. Defaults to None.
133
134        Returns:
135            np.ndarray: Gradients of the loss with respect to the module's
136                input.
137
138        Raises:
139            ValueError: If a classical layer is reached without gradients.
140
141        Examples:
142            >>> # Backward pass from loss
143            >>> dLdout = loss_fn.grads()
144            >>> grad = model.backward(dLdout)
145        """
146        for net in self._nets[::-1]:
147            if net.trainable:
148                if not isinstance(net, Ansatz):
149                    if dLdout is None:
150                        raise ValueError("Classical layers must pass in gradients.")
151                dLdout = net.backward(dLdout)
152        return dLdout
153
154    def random_init(self) -> None:
155        """
156        Randomly initialize all parameters in the module.
157
158        This method initializes parameters randomly for both layers and
159        ansätze. For ansätze, parameters are drawn from a standard normal
160        distribution. For layers, the default initialization method is used.
161
162        Examples:
163            >>> model.random_init()
164        """
165        for net in self._nets:
166            if isinstance(net, Ansatz):
167                keys = net.symbols
168                values = np.random.randn(len(net.symbols))
169                bindings = dict(zip(keys, values))
170                net.assign_parameters(bindings)
171            else:
172                net.init_params()

221    def save_checkpoint(self, model_path: str, ep: int, it: int, latest: bool = False) -> None:
222        """
223        Save the model state as a checkpoint.
224
225        This method saves the current state of all components, including
226        parameters and optimizer states, to disk.
227
228        Args:
229            model_path (str): Directory path for saving the checkpoint.
230            ep (int): Current epoch number.
231            it (int): Current iteration number.
232            latest (bool, optional): Whether this is the latest checkpoint.
233                If True, saves as model.npy. Defaults to False.
234
235        Examples:
236            >>> model.save_checkpoint("./checkpoints", ep=10, it=100)
237            >>> model.save_checkpoint("./checkpoints", ep=10, it=100, latest=True)
238        """
239        checkpoint = dict()
240        module_info = dict()
241        for i in range(len(self._nets)):
242            if isinstance(self._nets[i], Ansatz):
243                key = "ansatz{}".format(i)
244            else:
245                key = "layer{}".format(i)
246            module_info[key] = self._nets[i].summary
247        checkpoint["epoch"] = ep
248        checkpoint["iter"] = it
249        checkpoint["module"] = module_info
250
251        os.makedirs(model_path, exist_ok=True)
252        np.save("{}/model.npy".format(model_path), checkpoint)
253        if not latest:
254            shutil.copy(
255                "{0}/model.npy".format(model_path),
256                "{0}/{1}_{2}.npy".format(model_path, ep, it),
257            )

259    def load_checkpoint(self, model_path: str) -> tuple:
260        """
261        Load a model checkpoint from disk.
262
263        Args:
264            model_path (str): Path to the checkpoint file or directory.
265
266        Returns:
267            tuple: (epoch, iteration) where:
268                - epoch (int): Restored epoch number
269                - iteration (int): Restored iteration number + 1
270
271        Raises:
272            ValueError: If the checkpoint file cannot be found or loaded.
273
274        Examples:
275            >>> ep, it = model.load_checkpoint("./checkpoints/model.npy")
276            >>> ep, it = model.load_checkpoint("./checkpoints")
277        """
278
279        def find_fname(model_path):
280            f_list = sorted(os.listdir(model_path))
281            ep = 0
282            it = 0
283            for f in f_list:
284                if f.endswith(".npy"):
285                    nums = f[:-4].split("_")
286                    if len(nums) >= 2:
287                        ep = max(ep, int(nums[0]))
288                        it = max(it, int(nums[1]))
289            fname = str(ep) + "_" + str(it) + ".npy"
290            return fname
291
292        try:
293            if model_path.endswith(".npy"):
294                fname = model_path
295                checkpoint = np.load(fname, allow_pickle=True)
296            else:
297                if os.path.exists("{0}/model.npy".format(model_path)):
298                    fname = "model.npy"
299                else:
300                    fname = find_fname(model_path)
301                checkpoint = np.load("{0}/{1}".format(model_path, fname), allow_pickle=True)
302        except Exception as e:
303            raise ValueError(f"Invalid model path: {model_path}. Error: {e}")
304
305        try:
306            for val, net in zip(checkpoint.item()["module"].values(), self._nets):
307                net.load_params(val)
308        except Exception as e:
309            raise ValueError(f"Mismatched model. Error: {e}")
310
311        ep = checkpoint.item()["epoch"]
312        it = checkpoint.item()["iter"]
313        print(f"Successfully restored checkpoint at ep: {ep} it: {it}")
314        return ep, it + 1

327        """
328        Unfreeze all components in the module (enable training).

336    def _validate_nets(self, net_list: list) -> list:
337        """
338        Validate the components passed to the module.
339
340        Args:
341            net_list (list): List of components to validate.
342
343        Returns:
344            list: Validated component list.
345
346        Raises:
347            TypeError: If any component is not a Layer or Ansatz.
348            ValueError: If an Ansatz doesn't have measurements.
349
350        Examples:
351            >>> nets = self._validate_nets([ansatz, linear])
352        """
353        for net in net_list:
354            if not isinstance(net, (Layer, Ansatz)):
355                raise TypeError(f"Expected Layer or Ansatz, got {type(net).__name__}")
356            if isinstance(net, Ansatz):
357                if net.out_dim <= 0:
358                    raise ValueError("Ansatz must have measurements.")
359        return net_list
