cqlib_qml.data

Data loading and preprocessing utilities for quantum machine learning.

This module provides tools for loading, preprocessing, and batching data for quantum machine learning workflows. It includes dataset management, data loading, and preprocessing functions specifically designed for quantum image encoding.

Key Features: - Custom Dataset and DataLoader classes for quantum data - MNIST dataset preprocessing with quantum encoding - Image downscaling and grayscale conversion - Conflict removal for classification datasets - Integration with quantum encoders

Examples:

from cqlib_qml.data import Dataset, DataLoader from cqlib_qml.encoder import FRQI from cqlib_qml.data.data_preprocess import get_mnist_dataloader

Create a custom dataset

X = np.array([[1, 2], [3, 4], [5, 6]]) y = np.array([0, 1, 0]) dataset = Dataset(X, y)

Create a data loader

loader = DataLoader(dataset, batch_size=2, shuffle=True) for batch in loader: ... print(batch)

encoder = FRQI(n_pixels=16, grayscale=2) train_loader, test_loader = get_mnist_dataloader( ... classes=[0, 1], ... resize=(4, 4), ... encoding=encoder, ... batch_size=32 ... )

160
161    Attributes:
162        dataset (Dataset): The underlying dataset.

166

168        >>> from cqlib_qml.data import Dataset, DataLoader

205        """
206        Initialize a DataLoader instance.
207
208        Args:
209            dataset (Dataset): Dataset to load from.
210            batch_size (int, optional): Samples per batch. Defaults to 1.
211            shuffle (bool, optional): Shuffle data each epoch. Defaults to True.

213

215            >>> loader = DataLoader(dataset, batch_size=4, shuffle=False)
216        """
217        self._dataset = dataset
218        self._shuffle = shuffle
219        self._batch_size = batch_size
220        self._drop_last = drop_last
221
222        self._it = 0
223        self._end = self.__len__()
224        self._idx = np.arange(len(self._dataset))

226    def __len__(self) -> int:
227        """
228        Get the number of batches in the dataloader.
229
230        Returns:
231            int: Number of batches.
232
233        Examples:
234            >>> len(loader)
235            2
236        """
237        length = len(self._dataset)
238        length = length // self._batch_size if self._drop_last else np.ceil(length / self._batch_size)
239        return int(length)
240
241    def __iter__(self):
242        """
243        Return an iterator for the dataset.
244
245        Resets the iteration state and shuffles indices if enabled.
246
247        Returns:
248            DataLoader: The dataloader instance as an iterator.
249
250        Examples:
251            >>> for batch in loader:
252            ...     process(batch)
253        """
254        self._it = 0
255        if self._shuffle:
256            self._idx = np.arange(len(self._dataset))
257            np.random.shuffle(self._idx)
258        return self
259
260    def __next__(self):
261        """
262        Get the next batch of data.
263
264        Returns:
265            tuple: Batch data for the current iteration.
266
267        Raises:
268            StopIteration: When all batches have been iterated.
269
270        Examples:
271            >>> next(loader)  # Get first batch
272            (array([[6. , 5. ], [0.3, 7. ]]), array([0, 1]))
273        """
274        if self._it < self._end:
275            ret_data = self._dataset[self._idx[self._it * self._batch_size : (self._it + 1) * self._batch_size]]
276            self._it += 1
277            return ret_data
278        else:
279            raise StopIteration

def filter_targets(X: numpy.ndarray, Y: numpy.ndarray, classes: list) -> tuple: 
42def filter_targets(X: np.ndarray, Y: np.ndarray, classes: list) -> tuple:
43    """
44    Filter dataset to keep only specified classes.
45
46    This function selects samples that belong to the specified classes
47    and remaps labels to consecutive integers starting from 0.
48
49    Args:
50        X (np.ndarray): Input data array.
51        Y (np.ndarray): Label array.
52        classes (list): List of class labels to keep.
53
54    Returns:
55        tuple: (X_filtered, Y_filtered) where:
56            - X_filtered (np.ndarray): Filtered data array.
57            - Y_filtered (np.ndarray): Remapped label array.
58
59    Examples:
60        >>> # Keep only digits 0, 1, 2 from MNIST
61        >>> X, Y = filter_targets(X_train, Y_train, [0, 1, 2])
62        >>> print(np.unique(Y))
63        [0, 1, 2]
64    """
65    idx = Y == classes[0]
66    for i in range(1, len(classes)):
67        idx = idx | (Y == classes[i])
68    X, Y = (X[idx], Y[idx])
69    for i in range(len(Y)):
70        for j in range(len(classes)):
71            if Y[i] == classes[j]:
72                Y[i] = j
73    return X, Y

def remove_conflict(X: numpy.ndarray, Y: numpy.ndarray, resize: tuple) -> tuple: 
100def remove_conflict(X: np.ndarray, Y: np.ndarray, resize: tuple) -> tuple:
101    """
102    Remove samples with conflicting labels.
103
104    For each unique image, if it appears with multiple labels, all
105    conflicting samples are removed. This ensures label consistency.
106
107    Args:
108        X (np.ndarray): Image array.
109        Y (np.ndarray): Label array.
110        resize (tuple): Original image dimensions.
111
112    Returns:
113        tuple: (X_cleaned, Y_cleaned) with conflicting samples removed.
114
115    Examples:
116        >>> # Remove ambiguous samples
117        >>> X_clean, Y_clean = remove_conflict(X, Y, (4, 4))
118    """
119    x_dict = collections.defaultdict(set)
120    for x, y in zip(X, Y):
121        x_dict[tuple(x.numpy().flatten())].add(y.item())
122    X_rmcon = []
123    Y_rmcon = []
124    for x in x_dict.keys():
125        if len(x_dict[x]) == 1:
126            X_rmcon.append(np.array(x).reshape(resize))
127            Y_rmcon.append(list(x_dict[x])[0])
128    X = np.array(X_rmcon) / 255.0
129    Y = np.array(Y_rmcon)
130    return X, Y

def change_grayscale(X: numpy.ndarray, grayscale: int) -> numpy.ndarray: 
156def change_grayscale(X: np.ndarray, grayscale: int) -> np.ndarray:
157    """
158    Quantize images to a specified number of grayscale levels.
159
160    Args:
161        X (np.ndarray): Input image array in range [0, 1].
162        grayscale (int): Number of grayscale levels. Must be >= 2 and <= 256.
163
164    Returns:
165        np.ndarray: Quantized image array.
166
167    Raises:
168        AssertionError: If grayscale is not between 2 and 256.
169
170    Examples:
171        >>> # Reduce to 4 grayscale levels
172        >>> X_quantized = change_grayscale(X, 4)
173        >>> print(np.unique(X_quantized))
174        [0.0, 0.333, 0.667, 1.0]
175    """
176    if grayscale < 2 or grayscale > 256:
177        raise ValueError("grayscale should be between 2 and 256")
178
179    grays = np.linspace(0, 1, grayscale)
180    thresholds = np.linspace(0, 1, grayscale + 1)
181    for i in range(grayscale):
182        X[np.logical_and(X >= thresholds[i], X <= thresholds[i + 1])] = grays[i]
183    return X
