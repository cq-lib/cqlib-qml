cqlib_qml.encoder

Quantum encoding methods for classical data.

This module provides various quantum encoding strategies for mapping classical data onto quantum states. The choice of encoding method significantly affects the performance of quantum machine learning models.

Available Encoders: - AmplitudeEncoder: Encodes data into amplitudes of quantum states. - AngleEncoder: Encodes data as rotation angles of quantum gates. - BasisEncoder: Encodes integers as basis states. - FRQI: Flexible Representation of Quantum Images for image encoding. - NEQR: Novel Enhanced Quantum Representation for images. - QubitLattice: One-to-one mapping of binary pixels to qubits. - ZZFeatureEncoder: ZZFeatureMap style encoding with entanglement.

Image Encoders (FRQI, NEQR, QubitLattice) are specifically designed for encoding image data and support batch processing.

Examples:

from cqlib_qml.encoder import AmplitudeEncoder, AngleEncoder, FRQI

Amplitude encoding

amp_encoder = AmplitudeEncoder() circuits = amp_encoder(np.array([0.5, 0.3, 0.2]))

Angle encoding

ang_encoder = AngleEncoder(mode="classical") circuits = ang_encoder(np.array([0.5, 0.3]))

Image encoding with FRQI

frqi_encoder = FRQI(n_pixels=16, grayscale=2) img = np.random.rand(4, 4) circuit = frqi_encoder(img)

 74        """
 75        Encode data using amplitude encoding.
 76
 77        Args:
 78            data (np.ndarray): Input data array. Can be 1D (single sample)
 79                or 2D (multiple samples). Each sample will be normalized
 80                and padded to length 2^n.
 81
 82        Returns:
 83            list: List of Circuit objects with amplitude encoding.
 84                Each circuit represents one encoded sample.
 85
 86        Raises:
 87            ValueError: If the input vector is zero (norm = 0).
 88
 89        Examples:
 90            >>> # Single sample
 91            >>> circuits = encoder(np.array([0.5, 0.3]))
 92            >>>
 93            >>> # Multiple samples
 94            >>> data = np.array([[0.5, 0.3], [0.7, 0.2]])
 95            >>> circuits = encoder(data)
 96        """
 97        data = np.array([data]) if data.ndim == 1 else data
 98        enc_circs = []
 99        for vec in data:
100            norm = np.linalg.norm(vec)
101            if norm == 0:
102                raise ValueError("Cannot encode zero vector.")
103            vec = vec / np.linalg.norm(vec)
104            n_qubits = int(np.ceil(np.log2(len(vec))))
105            padded_vec = np.zeros(1 << n_qubits, dtype=np.complex128)
106            padded_vec[: len(vec)] = vec
107            circuit = Circuit(n_qubits)
108            self._build_recursive(padded_vec, list(range(n_qubits - 1, -1, -1)), circuit)
109            enc_circs.append(circuit)
110
111        return enc_circs

113    def _build_recursive(self, data: np.ndarray, qubits: list, circuit: Circuit) -> None:
114        """
115        Recursively build the amplitude encoding circuit.
116
117        Args:
118            data (np.ndarray): Normalized data vector.
119            qubits (list): List of qubit indices (from highest to lowest).
120            circuit (Circuit): Circuit to build.
121
122        Note:
123            This method modifies the circuit in-place.
124        """
125        n = len(qubits)
126        if n == 0 or len(data) == 1:
127            return
128
129        current_q = qubits[0]
130        remaining_q = qubits[1:]
131
132        half = len(data) // 2
133        left_norm = np.sqrt(sum(abs(data[i]) ** 2 for i in range(half)))
134        right_norm = np.sqrt(sum(abs(data[i]) ** 2 for i in range(half, len(data))))
135
136        total = np.sqrt(left_norm**2 + right_norm**2)
137        if total > 1e-10:
138            if left_norm > 0:
139                theta = 2 * np.arccos(left_norm / total)
140            else:
141                theta = np.pi
142        else:
143            theta = 0
144
145        circuit.ry(current_q, theta)
146
147        if left_norm > 1e-10 and half > 0:
148            left_data = data[:half] / left_norm
149            if len(remaining_q) > 0:
150                sub_cir = Circuit(len(remaining_q))
151                self._build_recursive(left_data, remaining_q, sub_cir)
152                for op in sub_cir.operations:
153                    instruction = op.instruction
154                    sub_qubits = [qid.index for qid in op.qubits]
155                    params = op.params
156                    if instruction.is_standard:
157                        gate = instruction.standard_gate
158                        cgate = McGate(1, gate)
159                        circuit.x(current_q)
160                        circuit.multi_control_gate(cgate, [current_q] + [remaining_q[i] for i in sub_qubits], params)
161                        circuit.x(current_q)
162                    elif instruction.is_mcgate:
163                        gate = instruction.mc_gate
164                        base_gate = gate.base_gate
165                        cgate = McGate(1 + gate.num_ctrl_qubits, base_gate)
166                        circuit.x(current_q)
167                        circuit.multi_control_gate(cgate, [current_q] + [remaining_q[i] for i in sub_qubits], params)
168                        circuit.x(current_q)
169
170        if right_norm > 1e-10 and half > 0:
171            right_data = data[half:] / right_norm
172            if len(remaining_q) > 0:
173                sub_cir = Circuit(len(remaining_q))
174                self._build_recursive(right_data, remaining_q, sub_cir)
175                for op in sub_cir.operations:
176                    instruction = op.instruction
177                    sub_qubits = [qid.index for qid in op.qubits]
178                    params = op.params
179                    if instruction.is_standard:
180                        gate = instruction.standard_gate
181                        cgate = McGate(1, gate)
182                        circuit.multi_control_gate(cgate, [current_q] + [remaining_q[i] for i in sub_qubits], params)
183                    elif instruction.is_mcgate:

185                        base_gate = gate.base_gate
186                        cgate = McGate(1 + gate.num_ctrl_qubits, base_gate)
187                        circuit.multi_control_gate(cgate, [current_q] + [remaining_q[i] for i in sub_qubits], params)

 33class AngleEncoder:
 34    """
 35    Angle encoder for quantum data representation.
 36
 37    Encodes classical data as rotation angles on quantum gates.
 38    Supports two modes:
 39        - classical: Each feature is encoded as RY(2*x) on a separate qubit
 40        - dense: Two features per qubit: RY(2*theta) and RZ(phi)
 41
 42    The factor of 2 in RY gates maps data from [0, 1] to [0, 2π],
 43    allowing full rotation range.
 44
 45    Args:
 46        mode (str): Encoding mode. Options: "classical" or "dense".
 47            Defaults to "classical".
 48
 49    Attributes:
 50        _mode (str): Current encoding mode.
 51
 52    Raises:
 53        ValueError: If mode is not "classical" or "dense".
 54
 55    Examples:
 56        >>> # Classical mode
 57        >>> encoder = AngleEncoder(mode="classical")
 58        >>> circuits = encoder(np.array([0.5, 0.3, 0.7]))
 59        >>>
 60        >>> # Dense mode
 61        >>> encoder = AngleEncoder(mode="dense")
 62        >>> circuits = encoder(np.array([0.5, 0.3, 0.7, 0.2]))
 63    """
 64
 65    __MODE = ["classical", "dense"]
 66
 67    def __init__(self, mode: str = "classical"):
 68        """
 69        Initialize an AngleEncoder instance.
 70
 71        Args:
 72            mode (str): Encoding mode. Must be "classical" or "dense".
 73                Defaults to "classical".
 74
 75        Raises:
 76            ValueError: If mode is not supported.
 77        """
 78        if mode not in AngleEncoder.__MODE:
 79            raise ValueError(f"Angle encoding only supports two modes: 'classical' and 'dense', " f"got '{mode}'.")
 80        self._mode = mode
 81
 82    def __call__(self, data: np.ndarray) -> list:
 83        """
 84        Encode data using angle encoding.
 85
 86        Args:
 87            data (np.ndarray): Input data array. Can be 1D (single sample)
 88                or 2D (multiple samples).
 89
 90        Returns:
 91            list: List of Circuit objects with angle encoding.
 92
 93        Examples:
 94            >>> # Single sample
 95            >>> circuits = encoder(np.array([0.5, 0.3]))
 96            >>>
 97            >>> # Multiple samples
 98            >>> data = np.array([[0.5, 0.3], [0.7, 0.2]])
 99            >>> circuits = encoder(data)
100        """
101        data = np.array([data]) if data.ndim == 1 else data
102        n_dim = len(data) if data.ndim == 1 else data.shape[1]
103        if self._mode == "classical":
104            enc_cirs = []
105            for vec in data:
106                circuit = Circuit(n_dim)
107                for i, x in enumerate(vec):
108                    circuit.ry(i, 2 * x)
109                enc_cirs.append(circuit)
110        else:  # dense mode
111            enc_cirs = []
112            for vec in data:
113                n_qubits = (n_dim + 1) // 2
114                circuit = Circuit(n_qubits)
115                for i in range(n_qubits):
116                    theta = vec[2 * i] if 2 * i < n_dim else 0
117                    phi = vec[2 * i + 1] if 2 * i + 1 < n_dim else 0
118                    circuit.ry(i, 2 * theta)
119                    circuit.rz(i, phi)
120                enc_cirs.append(circuit)
121
122        return enc_cirs

87            for i, bit in enumerate(bin_x):
88                if bit == "1":
89                    circuit.x(i)
90            enc_circs.append(circuit)

 45class FRQI(ImageEncoder):
 46    """
 47    FRQI (Flexible Representation of Quantum Images) encoder.
 48
 49    Encodes 2^n × 2^n images into quantum circuits using (2n + 1) qubits.
 50
 51    The encoding consists of:
 52        1. Initializing all position qubits to |+⟩ states
 53        2. Applying controlled rotations based on pixel values
 54
 55    Args:
 56        n_pixels (int): Total number of pixels in the image.
 57            Must be a perfect square and a power of 4 (4, 16, 64, ...).
 58        grayscale (int, optional): Number of grayscale levels.
 59            Defaults to 2.
 60
 61    Attributes:
 62        _n_pixels (int): Number of pixels.
 63        _n_pos_qubits (int): Number of position qubits (log2(n_pixels)).
 64        _n_color_qubits (int): Number of color qubits (always 1 for FRQI).
 65        _n_qubits (int): Total number of qubits.
 66        _grayscale (int): Number of grayscale levels.
 67        _q_state (list): Current state of position qubits.
 68
 69    Raises:
 70        ValueError: If n_pixels is not 2^n × 2^n.
 71
 72    Examples:
 73        >>> # For a 2x2 image (4 pixels)
 74        >>> encoder = FRQI(n_pixels=4, grayscale=2)
 75        >>> img = np.array([[0, 1], [1, 0]])
 76        >>> circuit = encoder(img)
 77        >>> print(circuit.num_qubits)  # 2n + 1 = 3
 78        3
 79    """
 80
 81    def __init__(self, n_pixels: int, grayscale: int = 2):
 82        """
 83        Initialize an FRQI encoder.
 84
 85        Args:
 86            n_pixels (int): Number of pixels. Must be 2^n × 2^n.
 87            grayscale (int): Number of grayscale levels. Defaults to 2.
 88
 89        Raises:
 90            ValueError: If n_pixels is not a perfect power of 4.
 91        """
 92        edge_len = int(np.sqrt(n_pixels))
 93        if not (edge_len**2 == n_pixels and 1 << int(np.log2(edge_len)) == edge_len):
 94            raise ValueError("Only support images with a resolution of 2^n x 2^n.")
 95        super(FRQI, self).__init__(n_pixels)
 96        self._n_channel = 1
 97        self._grayscale = grayscale
 98        self._n_color_qubits = 1
 99        self._n_pos_qubits = int(np.log2(n_pixels))
100        self._n_qubits = self._n_pos_qubits + self._n_color_qubits
101        self._q_state = [0] * self._n_pos_qubits
102
103    def __call__(self, imgs, use_qic: bool = False):
104        """
105        Encode images using FRQI.
106
107        Args:
108            imgs (Union[list, np.ndarray]): Input image(s). Can be a single
109                image (2D array) or a batch (3D or 4D array).
110            use_qic (bool, optional): Whether to use Quantum Image
111                Compression for optimization. Defaults to False.
112
113        Returns:
114            Union[Circuit, list]: Encoded quantum circuit(s). Returns a
115                single Circuit for one image, or a list for multiple images.
116
117        Raises:
118            ValueError: If image dimensions or values are invalid.
119
120        Examples:
121            >>> img = np.random.rand(4, 4)  # 4x4 image
122            >>> circuit = encoder(img)
123            >>>
124            >>> # With QIC optimization
125            >>> circuit = encoder(img, use_qic=True)
126        """
127        if not self._is_imgs_batch(imgs):
128            imgs = [imgs]
129
130        enc_cirs = []
131        for img in imgs:
132            img = self._img_preprocess(img, flatten=True)
133            encoder = self._construct_encoder(img, use_qic)
134            enc_cirs.append(encoder)
135
136        return enc_cirs[0] if len(enc_cirs) == 1 else enc_cirs

138    def __str__(self):
139        return "FRQI(n_qubits={}, grayscale={})".format(self._n_qubits, self._grayscale)

141    def _construct_encoder(self, img: np.ndarray, use_qic: bool) -> Circuit:
142        """
143        Construct the FRQI encoding circuit.
144
145        Args:
146            img (np.ndarray): Flattened image array.
147            use_qic (bool): Whether to use QIC optimization.
148
149        Returns:
150            Circuit: Encoded quantum circuit.
151        """
152        # Step 1: Prepare |H⟩ state (Hadamard on position qubits)
153        encoder = Circuit(self._n_qubits)
154        for qid in range(self._n_pos_qubits):
155            encoder.h(qid)
156
157        # Step 2: Apply rotation gates to encode pixel values
158        if use_qic:
159            frqi_circuit = self._construct_qic_circuit(img, rotate=True)
160        else:
161            frqi_circuit = self._construct_circuit(img, rotate=True)
162        encoder.compose(frqi_circuit)
163        return encoder

165    def _img_preprocess(self, img: np.ndarray, flatten: bool = True) -> np.ndarray:
166        """
167        Preprocess image for encoding.
168
169        Args:
170            img (np.ndarray): Input image.
171            flatten (bool): Whether to flatten the image. Defaults to True.
172
173        Returns:
174            np.ndarray: Preprocessed image.
175
176        Raises:
177            ValueError: If image is invalid.
178        """
179        self._validate_img(img)
180        if flatten:
181            img = img.flatten()
182        return img

184    def _validate_img(self, img: np.ndarray) -> None:
185        """
186        Validate image for FRQI encoding.
187
188        Args:
189            img (np.ndarray): Input image.
190
191        Raises:
192            ValueError: If image is not square, not 2^n × 2^n,
193                or contains invalid pixel values.
194        """
195        # Check shape
196        if not (img.shape[0] == img.shape[1] and 1 << int(np.log2(img.shape[0])) == img.shape[0]):
197            raise ValueError("Invalid image. The image size should be 2^n x 2^n.")
198
199        # Check pixel values
200        if not (img >= 0.0).all():
201            raise ValueError("Invalid image. The pixel values of the image must be non-negative.")
202
203        # Check grayscale level
204        if np.unique(img).shape[0] > self._grayscale:
205            raise ValueError("Invalid image. The input image must conform to grayscale level.")

207    def _construct_circuit(self, img: np.ndarray, rotate: bool) -> Circuit:
208        """
209        Construct the standard FRQI circuit.
210
211        Args:
212            img (np.ndarray): Flattened image.
213            rotate (bool): Whether to use rotation gates.
214
215        Returns:
216            Circuit: FRQI circuit.
217        """
218        circuit = Circuit(self._n_qubits)
219        for i in range(self._n_pixels):
220            if img[i] < 1e-12:
221                continue
222            # Set position qubits
223            bin_pos = bin(i)[2:].zfill(self._n_pos_qubits)
224            for qid in range(self._n_pos_qubits):
225                if (bin_pos[qid] == "0" and self._q_state[qid] == 0) or (
226                    bin_pos[qid] == "1" and self._q_state[qid] == 1
227                ):
228                    circuit.x(qid)
229                    self._q_state[qid] = 1 - self._q_state[qid]
230            # Apply color encoding
231            if rotate:
232                theta = float(img[i] / np.max(img) * np.pi)
233                cgate = McGate(self._n_pos_qubits, StandardGate.RY)
234                circuit.multi_control_gate(cgate, list(range(self._n_qubits)), [theta])
235            else:
236                pixel_val = int(img[i] / np.max(img) * 255)
237                bin_color = bin(pixel_val)[2:].zfill(self._n_color_qubits)
238                for qid in range(self._n_color_qubits):
239                    if bin_color[qid] == "1":
240                        mct_qids = list(range(self._n_pos_qubits)) + [self._n_pos_qubits + qid]
241                        cgate = McGate(self._n_pos_qubits, StandardGate.X)
242                        circuit.multi_control_gate(cgate, mct_qids)
243        # Reset position qubits
244        for qid in range(self._n_pos_qubits):
245            if self._q_state[qid] == 1:
246                circuit.x(qid)
247                self._q_state[qid] = 1 - self._q_state[qid]
248
249        return circuit
250
251    def _construct_qic_circuit(self, img: np.ndarray, rotate: bool, gid: int = 0) -> Circuit:
252        """
253        Construct FRQI circuit with QIC optimization.
254
255        Args:
256            img (np.ndarray): Flattened image.
257            rotate (bool): Whether to use rotation gates.
258            gid (int): Gate index. Defaults to 0.
259
260        Returns:
261            Circuit: Optimized FRQI circuit.
262        """
263        qic_circuit = Circuit(self._n_qubits)
264        img_dict = self._get_img_dict(img, bin_val=True)
265        for key in img_dict.keys():
266            theta = float(key) / np.max(img) * np.pi if rotate else None
267            min_dnf = self._get_min_expression(img_dict[key])
268            dnf_circuit = self._construct_dnf_circuit(min_dnf, gid, theta)
269            qic_circuit = qic_circuit + dnf_circuit
270        for qid in range(self._n_pos_qubits):
271            if self._q_state[qid] == 1:
272                qic_circuit.x(qid)
273                self._q_state[qid] = 1 - self._q_state[qid]
274
275        return qic_circuit
276
277    def _construct_dnf_circuit(self, min_dnf, gid: int = 0, theta: float = None) -> Circuit:
278        """
279        Construct DNF circuit for QIC.
280
281        Args:
282            min_dnf: Minimum DNF expression.
283            gid (int): Gate index.
284            theta (float, optional): Rotation angle.
285
286        Returns:
287            Circuit: DNF circuit.
288        """
289        dnf_circuit = Circuit(self._n_qubits)
290        cnf_list = self._split_dnf(min_dnf)
291        if cnf_list == ["True"]:
292            if theta is None:
293                dnf_circuit.x(gid + self._n_pos_qubits)
294            else:
295                dnf_circuit.ry(gid + self._n_pos_qubits, theta)
296            return dnf_circuit
297        for i in range(len(cnf_list)):
298            if i > 0:
299                uniqueness_dnf = self._get_uniqueness_dnf(cnf_list[:i], cnf_list[i])
300                uniqueness_dnf_circuit = self._construct_dnf_circuit(uniqueness_dnf, gid, theta)
301                dnf_circuit = dnf_circuit + uniqueness_dnf_circuit
302            else:
303                cnf_circuit = self._construct_cnf_circuit(cnf_list[i], gid=gid, theta=theta)
304                dnf_circuit = dnf_circuit + cnf_circuit
305
306        return dnf_circuit
307
308    def _construct_cnf_circuit(self, cnf, gid=0, theta=None) -> Circuit:
309        """
310        Construct CNF circuit for QIC.
311
312        Args:
313            cnf: CNF expression.
314            gid (int): Gate index.
315            theta (float, optional): Rotation angle.
316
317        Returns:
318            Circuit: CNF circuit.
319        """
320        cnf_circuit = Circuit(self._n_qubits)
321        items = self._split_cnf(cnf)
322        qids = self._get_cnf_qid(items)
323
324        for item, qid in zip(items, qids):
325            if (item[0] == "~" and self._q_state[qid] == 0) or (item[0] != "~" and self._q_state[qid] != 0):
326                cnf_circuit.x(qid)
327                self._q_state[qid] = 1 - self._q_state[qid]
328
329        if theta is None:
330            cgate = McGate(len(qids), StandardGate.X)
331            cnf_circuit.multi_control_gate(cgate, qids + [gid + self._n_pos_qubits])
332        else:
333            cgate = McGate(len(qids), StandardGate.RY)
334            cnf_circuit.multi_control_gate(cgate, qids + [gid + self._n_pos_qubits], [theta])
335        return cnf_circuit
336
337    def _get_uniqueness_dnf(self, pre_cnf_list, current_cnf):
338        """Get uniqueness DNF expression."""
339        uniqueness_dnf = ""
340        for cnf in pre_cnf_list:
341            uniqueness_dnf += "~(" + cnf + ") & "
342        uniqueness_dnf += "(" + current_cnf + ")"
343        uniqueness_dnf = to_dnf(uniqueness_dnf, simplify=True, force=True)
344        return uniqueness_dnf
345
346    def _get_cnf_qid(self, cnf_items):
347        """Extract qubit indices from CNF items."""
348        idx_list = []
349        for item in cnf_items:
350            idx_list.append(int(item[item.index("_") + 1 :]))
351        return idx_list
352
353    def _split_dnf(self, dnf):
354        """Split DNF expression into clauses."""
355        return str(dnf).replace("(", "").replace(")", "").split(" | ")
356
357    def _split_cnf(self, cnf):
358        """Split CNF expression into clauses."""
359        return str(cnf).replace("(", "").replace(")", "").split(" & ")
360
361    def _get_img_dict(self, img, bin_key=False, bin_val=False):
362        """Group pixels by color value."""
363        img_dict = dict()
364        for i in range(self._n_pixels):
365            if img[i] < 1e-12:
366                continue
367            key = bin(img[i])[2:].zfill(self._n_color_qubits) if bin_key else str(img[i])
368            val = bin(i)[2:].zfill(self._n_pos_qubits) if bin_val else i
369            if key not in img_dict.keys():
370                img_dict[key] = [val]
371            else:
372                img_dict[key].append(val)
373        return img_dict
374
375    def _get_boolen_expression(self, pixel):
376        """Get Boolean expression for a pixel position."""
377        boolen_expression = ""
378        x = symbols("x_0:" + str(self._n_pixels))
379        for i in range(self._n_pos_qubits):
380            boolen_expression += "~" + str(x[i]) + " " if pixel[i] == "0" else str(x[i]) + " "
381            if i != self._n_pos_qubits - 1:
382                boolen_expression += "& "
383        return "( " + boolen_expression + ")"
384
385    def _get_min_expression(self, pixels):
386        """Get minimum Boolean expression for pixels."""
387        boolen_expressions = ""
388        for i in range(len(pixels)):
389            boolen_expressions += self._get_boolen_expression(pixels[i])
390            if i != len(pixels) - 1:
391                boolen_expressions += " | "
392        min_expression = to_dnf(boolen_expressions, simplify=True, force=True)
393        return min_expression

The encoding consists of: 1. Initializing all position qubits to |+⟩ states 2. Applying controlled rotations based on pixel values

141        data = np.array([data]) if data.ndim == 1 else data
142        n_qubits = len(data) if data.ndim == 1 else data.shape[1]
143        enc_cirs = []
144        for vec in data:
145            circuit = Circuit(n_qubits)
146            for _ in range(self._n_repeats):
147                # Apply Hadamard gates
148                for i in range(n_qubits):
149                    circuit.h(i)
150                # Encode features as RZ rotations
151                for i in range(n_qubits):
152                    circuit.rz(i, vec[i] * 2 * np.pi)
153                # Apply entanglement
154                pairs = self._get_entanglement_pairs(n_qubits)
155                for i, j in pairs:
156                    circuit.rzz(i, j, (np.pi - vec[i]) * (np.pi - vec[j]))
157            enc_cirs.append(circuit)
158
159        return enc_cirs
