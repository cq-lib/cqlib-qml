cqlib_qml.optimizer

Optimization algorithms for training quantum models.

This module provides implementations of standard optimization algorithms with support for learning rate scheduling and gradient clipping.

All optimizers follow a common interface: 1. update: Compute the parameter update 2. step: Increment the step counter 3. reset_step: Reset the step counter

The optimizers maintain a cache for each parameter to store state information (e.g., momentum, running averages).

References: - SGD: Robbins & Monro (1951) - AdaGrad: Duchi et al. (2011) - RMSProp: Tieleman & Hinton (2012) - Adam: Kingma & Ba (2015)

Examples:

from cqlib_qml.optimizer import Adam, OptimizerInitializer

Create optimizer directly

opt = Adam(lr=0.001)

Create optimizer from string

opt = OptimizerInitializer("adam(lr=0.001)")()

Update parameters

param = np.array([1.0, 2.0]) grad = np.array([0.1, 0.2]) new_param = opt.update(param, grad, "weight")

 48class OptimizerBase(ABC):
 49    """
 50    Abstract base class for all optimizers.
 51
 52    This class defines the interface that all optimizers must implement.
 53    It provides common functionality for parameter updates, learning rate
 54    scheduling, and state management.
 55
 56    Attributes:
 57        cache (dict): Cache for optimizer state (momentum, running averages).
 58        cur_step (int): Current optimization step.
 59        hyperparameters (dict): Optimizer hyperparameters.
 60        lr (float): Learning rate.
 61        lr_scheduler (SchedulerBase): Learning rate scheduler.
 62
 63    Note:
 64        User-defined optimizers should inherit from this class and
 65        implement the `update` method.
 66
 67    Examples:
 68        >>> class MyOptimizer(OptimizerBase):
 69        ...     def update(self, param, param_grad, param_name, cur_loss=None):
 70        ...         return param - self.lr * param_grad
 71    """
 72
 73    def __init__(self, lr: float, scheduler=None):
 74        """
 75        Initialize an OptimizerBase instance.
 76
 77        Args:
 78            lr (float): Learning rate.
 79            scheduler (optional): Learning rate scheduler. Defaults to None.
 80        """
 81        self.cache = {}
 82        self.cur_step = 0
 83        self.hyperparameters = {}
 84        self.lr = lr
 85        self.lr_scheduler = SchedulerInitializer(scheduler, lr=lr)()
 86
 87    def __call__(
 88        self, param: np.ndarray, param_grad: np.ndarray, param_name: str, cur_loss: float = None
 89    ) -> np.ndarray:
 90        """
 91        Compute the parameter update.
 92
 93        This is a wrapper around the `update` method for convenient calling.
 94
 95        Args:
 96            param (np.ndarray): Current parameter values.
 97            param_grad (np.ndarray): Parameter gradients.
 98            param_name (str): Parameter name for state tracking.
 99            cur_loss (float, optional): Current loss for scheduling.
100                Defaults to None.
101
102        Returns:
103            np.ndarray: Updated parameter values.
104        """
105        return self.update(param, param_grad, param_name, cur_loss)

108        """Increment the optimizer step counter by 1."""
109        self.cur_step += 1

126        Set the learning rate scheduler.

139    def set_params(self, hparam_dict: dict = None, cache_dict: dict = None):
140        """
141        Set optimizer parameters from dictionaries.
142
143        Args:
144            hparam_dict (dict, optional): Hyperparameters dictionary.
145            cache_dict (dict, optional): Cache dictionary.
146
147        Returns:
148            OptimizerBase: The optimizer instance.
149        """
150        if hparam_dict is not None:
151            for k, v in hparam_dict.items():

155                        self.lr_scheduler = SchedulerInitializer(v, lr=None)()
156
157        if cache_dict is not None:
158            for k, v in cache_dict.items():
159                self.cache[k] = v
160        return self

182class OptimizerInitializer:
183    """
184    Factory class for initializing optimizers.
185
186    Provides a convenient way to create optimizer instances from strings,
187    dictionaries, or existing optimizer instances.
188
189    Args:
190        param: Optimizer configuration. Can be:
191            - None: Returns default SGD
192            - str: Optimizer name with optional parameters
193            - dict: Optimizer configuration dictionary
194            - OptimizerBase: Existing optimizer instance
195
196    Attributes:
197        param: The optimizer configuration.
198
199    Examples:
200        >>> # From string
201        >>> init = OptimizerInitializer("adam(lr=0.001)")
202        >>> opt = init()
203        >>>
204        >>> # From dict
205        >>> init = OptimizerInitializer({
206        ...     "hyperparameters": {"id": "Adam", "lr": 0.001},
207        ...     "cache": {}
208        ... })
209        >>> opt = init()
210        >>>
211        >>> # From existing optimizer
212        >>> init = OptimizerInitializer(Adam(lr=0.001))
213        >>> opt = init()
214    """
215
216    def __init__(self, param=None):
217        """
218        Initialize an OptimizerInitializer instance.
219
220        Args:
221            param: Optimizer configuration.
222        """
223        self.param = param
224
225    def __call__(self):
226        """
227        Create the optimizer instance.
228
229        Returns:
230            OptimizerBase: The optimizer instance.
231        """
232        param = self.param
233        if param is None:
234            opt = SGD()
235        elif isinstance(param, OptimizerBase):
236            opt = param
237        elif isinstance(param, str):
238            opt = self.init_from_str()
239        elif isinstance(param, dict):
240            opt = self.init_from_dict()
241        return opt
242
243    def init_from_str(self):
244        """
245        Initialize optimizer from a string.
246
247        Supported formats:
248            - "adam": Adam with default parameters
249            - "adam(lr=0.001)": Adam with custom learning rate
250            - "sgd(lr=0.01, momentum=0.9)": SGD with momentum
251
252        Returns:
253            OptimizerBase: The optimizer instance.
254
255        Raises:
256            ValueError: If optimizer name is not supported.
257        """
258        r = r"([a-zA-Z]*)=([^,)]*)"
259        opt_str = self.param.lower()
260        kwargs = dict([(i, eval(j)) for (i, j) in re.findall(r, opt_str)])
261        if "sgd" in opt_str:
262            optimizer = SGD(**kwargs)
263        elif "adagrad" in opt_str:
264            optimizer = AdaGrad(**kwargs)
265        elif "rmsprop" in opt_str:
266            optimizer = RMSProp(**kwargs)
267        elif "adam" in opt_str:
268            optimizer = Adam(**kwargs)
269        else:
270            raise ValueError(f"Unsupported optimizer: {opt_str}. " f"Supported: ['sgd', 'adagrad', 'rmsprop', 'adam']")
271        return optimizer
272
273    def init_from_dict(self):
274        """
275        Initialize optimizer from a dictionary.
276
277        Args:
278            param (dict): Configuration dictionary with 'hyperparameters'
279                and optionally 'cache' keys.
280
281        Returns:
282            OptimizerBase: The optimizer instance.
283
284        Raises:
285            ValueError: If optimizer ID is not supported.
286        """
287        O = self.param
288        cc = O["cache"] if "cache" in O else None
289        op = O["hyperparameters"] if "hyperparameters" in O else None
290
291        if op is None:
292            raise ValueError("Must have `hyperparameters` key.")
293
294        if op and op["id"] == "SGD":
295            optimizer = SGD().set_params(op, cc)
296        elif op and op["id"] == "RMSProp":
297            optimizer = RMSProp().set_params(op, cc)
298        elif op and op["id"] == "AdaGrad":
299            optimizer = AdaGrad().set_params(op, cc)
300        elif op and op["id"] == "Adam":
301            optimizer = Adam().set_params(op, cc)
302        elif op:
303            raise ValueError(f"Unsupported optimizer: {op['id']}. " f"Supported: ['SGD', 'RMSProp', 'AdaGrad', 'Adam']")
304        return optimizer

307class SGD(OptimizerBase):
308    """
309    Stochastic Gradient Descent optimizer with momentum.
310
311    update^t = momentum * update^{t-1} + lr * ∇L
312    θ^{t+1} = θ^t - update^t
313
314    Args:
315        lr (float, optional): Learning rate. Defaults to 0.01.
316        momentum (float, optional): Momentum coefficient in [0, 1].
317            Defaults to 0.
318        clip_norm (float, optional): Maximum L2 norm for gradients.
319            Defaults to None.
320        lr_scheduler (optional): Learning rate scheduler. Defaults to None.
321
322    Examples:
323        >>> # Basic SGD
324        >>> opt = SGD(lr=0.01)
325        >>>
326        >>> # SGD with momentum
327        >>> opt = SGD(lr=0.01, momentum=0.9)
328        >>>
329        >>> # SGD with gradient clipping
330        >>> opt = SGD(lr=0.01, clip_norm=1.0)
331    """
332
333    def __init__(self, lr=0.01, momentum=0.0, clip_norm=None, lr_scheduler=None, **kwargs):
334        """Initialize an SGD instance."""
335        super().__init__(lr, lr_scheduler)
336
337        self.hyperparameters = {
338            "id": "SGD",
339            "lr": lr,
340            "momentum": momentum,
341            "clip_norm": clip_norm,
342            "lr_scheduler": str(self.lr_scheduler),
343        }
344
345    def __str__(self):
346        H = self.hyperparameters
347        return "SGD(lr={}, momentum={}, clip_norm={}, lr_scheduler={})".format(
348            H["lr"], H["momentum"], H["clip_norm"], H["lr_scheduler"]
349        )
350
351    def update(self, param: np.ndarray, param_grad: np.ndarray, param_name: str, cur_loss: float = None) -> np.ndarray:
352        """
353        Compute the SGD update.
354
355        Args:
356            param (np.ndarray): Current parameter values.
357            param_grad (np.ndarray): Parameter gradients.
358            param_name (str): Parameter name for state tracking.
359            cur_loss (float, optional): Current loss for scheduling.
360
361        Returns:
362            np.ndarray: Updated parameter values.
363        """
364        C = self.cache
365        H = self.hyperparameters
366        momentum, clip_norm = H["momentum"], H["clip_norm"]
367        lr = self.lr_scheduler(self.cur_step, cur_loss)
368
369        if param_name not in C:
370            C[param_name] = np.zeros_like(param_grad)
371
372        # Gradient clipping
373        t = np.inf if clip_norm is None else clip_norm
374        if norm(param_grad) > t:
375            param_grad = param_grad * t / norm(param_grad)
376
377        update = momentum * C[param_name] + lr * param_grad
378        self.cache[param_name] = update
379        return param - update

382class AdaGrad(OptimizerBase):
383    """
384    AdaGrad optimizer with adaptive learning rates.
385
386    cache[t] = cache[t-1] + grad[t]²
387    update[t] = lr * grad[t] / (√cache[t] + eps)
388    θ[t+1] = θ[t] - update[t]
389
390    References:
391        - Duchi, J., et al. (2011). "Adaptive Subgradient Methods for
392          Online Learning and Stochastic Optimization."
393
394    Args:
395        lr (float, optional): Learning rate. Defaults to 0.01.
396        eps (float, optional): Smoothing term. Defaults to 1e-7.
397        clip_norm (float, optional): Maximum L2 norm for gradients.
398            Defaults to None.
399        lr_scheduler (optional): Learning rate scheduler. Defaults to None.
400
401    Examples:
402        >>> opt = AdaGrad(lr=0.01)
403        >>> opt = AdaGrad(lr=0.01, eps=1e-8)
404    """
405
406    def __init__(self, lr=0.01, eps=1e-7, clip_norm=None, lr_scheduler=None, **kwargs):
407        """Initialize an AdaGrad instance."""
408        super().__init__(lr, lr_scheduler)
409
410        self.cache = {}
411        self.hyperparameters = {
412            "id": "AdaGrad",
413            "lr": lr,
414            "eps": eps,
415            "clip_norm": clip_norm,
416            "lr_scheduler": str(self.lr_scheduler),
417        }
418
419    def __str__(self):
420        H = self.hyperparameters
421        return "AdaGrad(lr={}, eps={}, clip_norm={}, lr_scheduler={})".format(
422            H["lr"], H["eps"], H["clip_norm"], H["lr_scheduler"]
423        )
424
425    def update(self, param: np.ndarray, param_grad: np.ndarray, param_name: str, cur_loss: float = None) -> np.ndarray:
426        """
427        Compute the AdaGrad update.
428
429        Args:
430            param (np.ndarray): Current parameter values.
431            param_grad (np.ndarray): Parameter gradients.
432            param_name (str): Parameter name for state tracking.
433            cur_loss (float, optional): Current loss for scheduling.
434
435        Returns:
436            np.ndarray: Updated parameter values.
437        """
438        C = self.cache
439        H = self.hyperparameters
440        eps, clip_norm = H["eps"], H["clip_norm"]
441        lr = self.lr_scheduler(self.cur_step, cur_loss)
442
443        if param_name not in C:
444            C[param_name] = np.zeros_like(param_grad)
445
446        # Gradient clipping
447        t = np.inf if clip_norm is None else clip_norm
448        if norm(param_grad) > t:
449            param_grad = param_grad * t / norm(param_grad)
450
451        C[param_name] += param_grad**2
452        update = lr * param_grad / (np.sqrt(C[param_name]) + eps)
453        self.cache = C
454        return param - update

457class RMSProp(OptimizerBase):
458    """
459    RMSProp optimizer with adaptive learning rates.
460
461    cache[t] = decay * cache[t-1] + (1-decay) * grad[t]²
462    update[t] = lr * grad[t] / (√cache[t] + eps)
463    θ[t+1] = θ[t] - update[t]
464
465    References:
466        - Tieleman, T., & Hinton, G. (2012). "RMSProp: Divide the
467          gradient by a running average of its recent magnitude."
468
469    Args:
470        lr (float, optional): Learning rate. Defaults to 0.001.
471        decay (float, optional): Decay rate in [0, 1]. Defaults to 0.9.
472        eps (float, optional): Smoothing term. Defaults to 1e-7.
473        clip_norm (float, optional): Maximum L2 norm for gradients.
474            Defaults to None.
475        lr_scheduler (optional): Learning rate scheduler. Defaults to None.
476
477    Examples:
478        >>> opt = RMSProp(lr=0.001, decay=0.9)
479        >>> opt = RMSProp(lr=0.001, decay=0.99, eps=1e-8)
480    """
481
482    def __init__(self, lr=0.001, decay=0.9, eps=1e-7, clip_norm=None, lr_scheduler=None, **kwargs):
483        """Initialize an RMSProp instance."""
484        super().__init__(lr, lr_scheduler)
485
486        self.cache = {}
487        self.hyperparameters = {
488            "id": "RMSProp",
489            "lr": lr,
490            "eps": eps,
491            "decay": decay,
492            "clip_norm": clip_norm,
493            "lr_scheduler": str(self.lr_scheduler),
494        }
495
496    def __str__(self):
497        H = self.hyperparameters
498        return "RMSProp(lr={}, eps={}, decay={}, clip_norm={}, lr_scheduler={})".format(
499            H["lr"], H["eps"], H["decay"], H["clip_norm"], H["lr_scheduler"]
500        )
501
502    def update(self, param: np.ndarray, param_grad: np.ndarray, param_name: str, cur_loss: float = None) -> np.ndarray:
503        """
504        Compute the RMSProp update.
505
506        Args:
507            param (np.ndarray): Current parameter values.
508            param_grad (np.ndarray): Parameter gradients.
509            param_name (str): Parameter name for state tracking.
510            cur_loss (float, optional): Current loss for scheduling.
511
512        Returns:
513            np.ndarray: Updated parameter values.
514        """
515        C = self.cache
516        H = self.hyperparameters
517        eps, decay, clip_norm = H["eps"], H["decay"], H["clip_norm"]
518        lr = self.lr_scheduler(self.cur_step, cur_loss)
519
520        if param_name not in C:
521            C[param_name] = np.zeros_like(param_grad)
522
523        # Gradient clipping
524        t = np.inf if clip_norm is None else clip_norm
525        if norm(param_grad) > t:
526            param_grad = param_grad * t / norm(param_grad)
527
528        C[param_name] = decay * C[param_name] + (1 - decay) * param_grad**2
529        update = lr * param_grad / (np.sqrt(C[param_name]) + eps)
530        self.cache = C
531        return param - update

534class Adam(OptimizerBase):
535    """
536    Adam (Adaptive Moment Estimation) optimizer.
537
538    m[t] = β1 * m[t-1] + (1-β1) * grad[t]  (First moment)
539    v[t] = β2 * v[t-1] + (1-β2) * grad[t]² (Second moment)
540    m_hat = m[t] / (1-β1^t)                 (Bias correction)
541    v_hat = v[t] / (1-β2^t)                 (Bias correction)
542    update = lr * m_hat / (√v_hat + eps)
543    θ[t+1] = θ[t] - update
544
545    References:
546        - Kingma, D. P., & Ba, J. (2015). "Adam: A Method for
547          Stochastic Optimization."
548
549    Args:
550        lr (float, optional): Learning rate. Defaults to 0.001.
551        decay1 (float, optional): Decay rate for first moment (β1).
552            Defaults to 0.9.
553        decay2 (float, optional): Decay rate for second moment (β2).
554            Defaults to 0.999.
555        eps (float, optional): Smoothing term. Defaults to 1e-7.
556        clip_norm (float, optional): Maximum L2 norm for gradients.
557            Defaults to None.
558        lr_scheduler (optional): Learning rate scheduler. Defaults to None.
559
560    Examples:
561        >>> # Default Adam
562        >>> opt = Adam()
563        >>>
564        >>> # Custom parameters
565        >>> opt = Adam(lr=0.001, decay1=0.9, decay2=0.999)
566        >>>
567        >>> # With gradient clipping
568        >>> opt = Adam(lr=0.001, clip_norm=1.0)
569    """
570
571    def __init__(
572        self,
573        lr=0.001,
574        decay1=0.9,
575        decay2=0.999,
576        eps=1e-7,
577        clip_norm=None,
578        lr_scheduler=None,
579        **kwargs,
580    ):
581        """Initialize an Adam instance."""
582        super().__init__(lr, lr_scheduler)
583
584        self.cache = {}
585        self.hyperparameters = {
586            "id": "Adam",
587            "lr": lr,
588            "eps": eps,
589            "decay1": decay1,
590            "decay2": decay2,
591            "clip_norm": clip_norm,
592            "lr_scheduler": str(self.lr_scheduler),
593        }
594
595    def __str__(self):
596        H = self.hyperparameters
597        return "Adam(lr={}, decay1={}, decay2={}, eps={}, clip_norm={}, lr_scheduler={})".format(
598            H["lr"], H["decay1"], H["decay2"], H["eps"], H["clip_norm"], H["lr_scheduler"]
599        )
600
601    def update(self, param: np.ndarray, param_grad: np.ndarray, param_name: str, cur_loss: float = None) -> np.ndarray:
602        """
603        Compute the Adam update.
604
605        Args:
606            param (np.ndarray): Current parameter values.
607            param_grad (np.ndarray): Parameter gradients.
608            param_name (str): Parameter name for state tracking.
609            cur_loss (float, optional): Current loss for scheduling.
610
611        Returns:
612            np.ndarray: Updated parameter values.
613        """
614        C = self.cache
615        H = self.hyperparameters
616        d1, d2 = H["decay1"], H["decay2"]
617        eps, clip_norm = H["eps"], H["clip_norm"]
618        lr = self.lr_scheduler(self.cur_step, cur_loss)
619
620        if param_name not in C:
621            C[param_name] = {
622                "t": 0,
623                "mean": np.zeros_like(param_grad),
624                "var": np.zeros_like(param_grad),
625            }
626
627        # Gradient clipping
628        t = np.inf if clip_norm is None else clip_norm
629        if norm(param_grad) > t:
630            param_grad = param_grad * t / norm(param_grad)
631
632        t = C[param_name]["t"] + 1
633        var = C[param_name]["var"]
634        mean = C[param_name]["mean"]
635
636        # Update cache
637        C[param_name]["t"] = t
638        C[param_name]["var"] = d2 * var + (1 - d2) * param_grad**2
639        C[param_name]["mean"] = d1 * mean + (1 - d1) * param_grad
640        self.cache = C
641
642        # Unbiased moment estimates
643        v_hat = C[param_name]["var"] / (1 - d2**t)
644        m_hat = C[param_name]["mean"] / (1 - d1**t)

536    """
537    Adam (Adaptive Moment Estimation) optimizer.
538
539    m[t] = β1 * m[t-1] + (1-β1) * grad[t]  (First moment)
540    v[t] = β2 * v[t-1] + (1-β2) * grad[t]² (Second moment)
541    m_hat = m[t] / (1-β1^t)                 (Bias correction)
542    v_hat = v[t] / (1-β2^t)                 (Bias correction)
543    update = lr * m_hat / (√v_hat + eps)
544    θ[t+1] = θ[t] - update
545
546    References:
547        - Kingma, D. P., & Ba, J. (2015). "Adam: A Method for
548          Stochastic Optimization."
