cqlib_qml.loss

Commonly used loss functions for quantum machine learning.

This module provides implementations of standard loss functions with automatic differentiation support via autograd.

All loss functions inherit from the LossFun base class and implement: 1. Forward computation: __call__ / _get_loss 2. Gradient computation: grads method

The gradients are computed using autograd's automatic differentiation, which provides exact gradients for the implemented loss functions.

Examples:

from cqlib_qml.loss import MSELoss, SoftmaxCrossEntropy

Using MSE loss

mse = MSELoss() loss = mse(predictions, targets) gradients = mse.grads()

Using Softmax Cross Entropy

ce = SoftmaxCrossEntropy() loss = ce(logits, labels) gradients = ce.grads()

 35class LossFun(ABC):
 36    """
 37    Base class for all loss functions.
 38
 39    This abstract class defines the interface that all loss functions
 40    must implement. It provides the common structure for loss computation
 41    and gradient calculation.
 42
 43    Attributes:
 44        _pred (np.ndarray): Stored predictions from the last call.
 45        _target (np.ndarray): Stored targets from the last call.
 46
 47    Note:
 48        User-defined loss functions should inherit from this class and
 49        implement the `_get_loss` method.
 50
 51    Examples:
 52        >>> class MyLoss(LossFun):
 53        ...     def _get_loss(self, pred, target):
 54        ...         return np.mean((pred - target) ** 2)
 55        >>>
 56        >>> loss_fn = MyLoss()
 57        >>> loss = loss_fn(pred, target)
 58        >>> grads = loss_fn.grads()
 59    """
 60
 61    def __init__(self):
 62        """Initialize a LossFun instance."""
 63        self._pred = None
 64        self._target = None
 65
 66    def __call__(self, pred: np.ndarray, target: np.ndarray):
 67        """
 68        Compute the loss value.
 69
 70        Args:
 71            pred (np.ndarray): Predicted values.
 72            target (np.ndarray): Ground truth values.
 73
 74        Returns:
 75            float: Loss value.
 76
 77        Raises:
 78            ValueError: If predictions and targets have different shapes.
 79
 80        Examples:
 81            >>> loss = loss_fn(pred, target)
 82        """
 83        if pred.shape != target.shape:
 84            raise ValueError(
 85                f"Predicted values shape {pred.shape} and ground truth " f"shape {target.shape} must be the same."
 86            )
 87        self._pred = pred
 88        self._target = target
 89        loss = self._get_loss(pred, target)
 90        return loss
 91
 92    @abstractmethod
 93    def _get_loss(self, pred: np.ndarray, target: np.ndarray):
 94        """
 95        Compute the loss value (to be implemented by subclasses).
 96
 97        Args:
 98            pred (np.ndarray): Predicted values.
 99            target (np.ndarray): Ground truth values.
100
101        Returns:
102            float: Loss value.
103
104        Raises:
105            NotImplementedError: If not implemented by subclass.
106        """
107        raise NotImplementedError
108
109    def grads(self, dpred: np.ndarray = None) -> np.ndarray:
110        """
111        Compute gradients of the loss with respect to predictions.
112
113        This method uses autograd to compute exact gradients. Optionally,
114        gradients can be scaled by dpred for chain rule propagation.
115
116        Args:
117            dpred (np.ndarray, optional): Gradients from subsequent layers.
118                Defaults to None.
119
120        Returns:
121            np.ndarray: Gradients of the loss with respect to predictions.
122
123        Examples:
124            >>> # Get raw gradients
125            >>> grads = loss_fn.grads()
126            >>>
127            >>> # Scale gradients for chain rule
128            >>> grads = loss_fn.grads(dpred=2.0)
129        """
130        fun_grad = grad(self._get_loss)
131        gradients = fun_grad(self._pred, self._target)
132        gradients[abs(self._pred - self._target) < 1e-12] = 0
133        if dpred is not None:
134            gradients *= dpred
135        return gradients

138class HingeLoss(LossFun):
139    """
140    Hinge loss for classification.
141
142    L_Hinge(y) = max(0, 1 - ŷ * y)
143
144    The hinge loss is commonly used for SVM-like classifiers. It penalizes
145    predictions that are on the wrong side of the margin.
146
147    Args:
148        None
149
150    References:
151        - Cortes, C., & Vapnik, V. (1995). "Support-vector networks."
152
153    Examples:
154        >>> hinge = HingeLoss()
155        >>> pred = np.array([0.8, -0.2])
156        >>> target = np.array([1, -1])
157        >>> loss = hinge(pred, target)
158        >>> grads = hinge.grads()
159    """
160
161    def __init__(self):
162        """Initialize a HingeLoss instance."""
163        super().__init__()
164
165    def _get_loss(self, pred: np.ndarray, target: np.ndarray):
166        """
167        Compute the hinge loss.
168
169        Args:
170            pred (np.ndarray): Predicted values.
171            target (np.ndarray): Target values in {-1, 1}.
172
173        Returns:
174            float: Hinge loss value.
175        """
176        loss = np.clip(1 - pred * target, a_min=0.0, a_max=None)
177        return np.mean(loss)
178
179    def __str__(self):
180        return "HingeLoss"

183class MSELoss(LossFun):
184    """
185    Mean Squared Error loss for regression.
186
187    L_MSE(y) = (1/N) * Σ(ŷ_i - y_i)²
188
189    The MSE loss is commonly used for regression tasks. It penalizes
190    large errors more heavily than small errors.
191
192    Args:
193        None
194
195    Examples:
196        >>> mse = MSELoss()
197        >>> pred = np.array([0.5, 0.3, 0.7])
198        >>> target = np.array([0.5, 0.3, 0.7])
199        >>> loss = mse(pred, target)  # Returns 0.0
200        >>>
201        >>> pred = np.array([0.6, 0.4])
202        >>> target = np.array([0.5, 0.3])
203        >>> loss = mse(pred, target)  # Returns 0.01
204    """
205
206    def __init__(self):
207        """Initialize an MSELoss instance."""
208        super().__init__()
209
210    def _get_loss(self, pred: np.ndarray, target: np.ndarray):
211        """
212        Compute the mean squared error loss.
213
214        Args:
215            pred (np.ndarray): Predicted values.
216            target (np.ndarray): Ground truth values.
217
218        Returns:
219            float: MSE loss value.
220        """
221        loss = (pred - target) ** 2
222        return np.mean(loss)
223
224    def __str__(self):
225        return "MSELoss"

228class BCELoss(LossFun):
229    """
230    Binary Cross Entropy loss for binary classification.
231
232    L_BCE(y) = -(1/N) * (y · log(ŷ) + (1-y) · log(1-ŷ))
233
234    The BCE loss is used for binary classification tasks where targets
235    are in {0, 1}.
236
237    Args:
238        None
239
240    Note:
241        Targets should be in the range [0, 1].
242
243    Examples:
244        >>> bce = BCELoss()
245        >>> pred = np.array([0.9, 0.1])
246        >>> target = np.array([1.0, 0.0])
247        >>> loss = bce(pred, target)
248        >>> grads = bce.grads()
249    """
250
251    def __init__(self):
252        """Initialize a BCELoss instance."""
253        super().__init__()
254
255    def _get_loss(self, pred: np.ndarray, target: np.ndarray):
256        """
257        Compute the binary cross entropy loss.
258
259        Args:
260            pred (np.ndarray): Predicted probabilities in (0, 1).
261            target (np.ndarray): Target values in {0, 1}.
262
263        Returns:
264            float: BCE loss value.
265        """
266        eps = np.finfo(float).eps
267        loss = np.clip(-target * np.log(pred + eps) - (1 - target) * np.log(1 - pred + eps), 0, 100)
268        return np.mean(loss)
269
270    def __str__(self):
271        return "BCELoss"

274class CrossEntropy(LossFun):
275    """
276    Cross Entropy loss for multi-class classification.
277
278    L_CE(y) = -Σ y_i · log(ŷ_i)
279
280    This loss is used for multi-class classification with one-hot encoded
281    targets. Predictions should be probabilities that sum to 1.
282
283    Args:
284        None
285
286    Note:
287        Targets should be one-hot encoded. Predictions should be
288        probabilities (e.g., from softmax activation).
289
290    Examples:
291        >>> ce = CrossEntropy()
292        >>> pred = np.array([[0.7, 0.2, 0.1]])
293        >>> target = np.array([[1.0, 0.0, 0.0]])
294        >>> loss = ce(pred, target)
295        >>> grads = ce.grads()
296    """
297
298    def __init__(self):
299        """Initialize a CrossEntropyLoss instance."""
300        super().__init__()
301
302    def _get_loss(self, pred: np.ndarray, target: np.ndarray):
303        """
304        Compute the cross entropy loss.
305
306        Args:
307            pred (np.ndarray): Predicted probabilities (sum to 1 per sample).
308            target (np.ndarray): One-hot encoded target values.
309
310        Returns:
311            float: Cross entropy loss value.
312        """
313        eps = np.finfo(float).eps
314        loss = np.clip(-np.sum(target * np.log(pred + eps)), 0, 100)
315        return loss
316
317    def __str__(self):
318        return "CrossEntropyLoss"

321class SoftmaxCrossEntropy(LossFun):
322    """
323    Cross Entropy loss with built-in softmax activation.
324
325    This loss combines softmax activation with cross entropy loss.
326    It takes raw logits as input and applies softmax internally.
327
328    L_CE(y) = -Σ y_i · log(softmax(ŷ)_i)
329
330    Args:
331        None
332
333    Note:
334        This loss should be used when the last layer produces raw logits
335        (without softmax activation). The softmax is applied internally.
336
337    Examples:
338        >>> ce = SoftmaxCrossEntropy()
339        >>> logits = np.array([[1.0, 0.5, 0.2]])
340        >>> target = np.array([[1.0, 0.0, 0.0]])
341        >>> loss = ce(logits, target)
342        >>> grads = ce.grads()
343        >>>
344        >>> # Gradients are with respect to logits
345        >>> print(grads.shape)
346        (1, 3)
347    """
348
349    def __init__(self):
350        """Initialize a SoftmaxCrossEntropy instance."""
351        super().__init__()
352
353    def _get_loss(self, pred: np.ndarray, target: np.ndarray):
354        """
355        Compute the softmax cross entropy loss.
356
357        Args:
358            pred (np.ndarray): Raw logits.
359            target (np.ndarray): One-hot encoded target values.
360
361        Returns:
362            float: Cross entropy loss value.
363        """
364        exps = np.exp(pred)
365        self._pred = exps / np.sum(exps, axis=1).reshape(-1, 1)
366        eps = np.finfo(float).eps
367        loss = np.clip(-np.sum(target * np.log(self._pred + eps)), 0, 100)
368        return loss
369
370    def grads(self, dpred: np.ndarray = None) -> np.ndarray:
371        """
372        Compute gradients of the loss with respect to logits.
373
374        For softmax cross entropy, the gradient simplifies to:
375        dL/dlogits = softmax(logits) - targets
376
377        Args:
378            dpred (np.ndarray, optional): Gradients from subsequent layers.
379                Defaults to None.
380
381        Returns:
382            np.ndarray: Gradients with respect to logits.
383
384        Examples:
385            >>> ce = SoftmaxCrossEntropy()
386            >>> logits = np.array([[1.0, 0.5, 0.2]])
387            >>> target = np.array([[1.0, 0.0, 0.0]])
388            >>> loss = ce(logits, target)
389            >>> grads = ce.grads()
390            >>> print(grads)
391            [[-0.09  0.12  0.09]]  # Approximate values
392        """
393        gradients = self._pred - self._target
394        gradients[abs(self._pred - self._target) < 1e-12] = 0
395        if dpred is not None:
396            gradients *= dpred
397        return gradients
398
399    def __str__(self):
400        return "CrossEntropyLoss"
