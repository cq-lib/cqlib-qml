cqlib_qml.layer

Classical neural network layers for hybrid quantum-classical models.

This module provides classical neural network layers that can be combined with quantum circuits (ansätze) to build hybrid quantum-classical models. These layers support automatic differentiation and are designed to be compatible with the quantum components.

Available Layers: - Layer: Base class for all neural network layers. - Linear: Fully connected (dense) layer with optional activation. - Activation: Activation functions for neural networks.

The layers support: - Automatic differentiation (backpropagation) - Parameter optimization - Checkpoint saving and loading - Freezing/unfreezing of trainable parameters

Examples:

from cqlib_qml.layer import Linear, ActivationInitializer

Create a linear layer with sigmoid activation

layer = Linear(in_dim=10, out_dim=5, act_fn="sigmoid")

Forward pass

X = np.random.randn(32, 10) output = layer.forward(X)

Backward pass

dLdy = np.random.randn(32, 5) dX = layer.backward(dLdy)

class Linear(cqlib_qml.layer.Layer): 
 37class Linear(Layer):
 38    """
 39    Fully connected linear layer.
 40
 41    Computes y = xA^T + b, where A is the weight matrix and b is the bias.
 42    An optional activation function can be applied after the linear transformation.
 43
 44    Args:
 45        in_dim (int): Dimension of the input features.
 46        out_dim (int): Dimension of the output features.
 47        bias (bool, optional): Whether to include a bias term.
 48            Defaults to True.
 49        act_fn (str, optional): Activation function to apply.
 50            Options: "sigmoid", "relu", "tanh", "softplus", None.
 51            Defaults to None.
 52
 53    Attributes:
 54        in_dim (int): Input dimension.
 55        out_dim (int): Output dimension.
 56        hyperparameters (dict): Layer hyperparameters.
 57
 58    Raises:
 59        ValueError: If input dimensions don't match during forward/backward.
 60
 61    Examples:
 62        >>> # Linear layer with sigmoid activation
 63        >>> layer = Linear(in_dim=10, out_dim=5, act_fn="sigmoid")
 64        >>>
 65        >>> # Linear layer without bias
 66        >>> layer = Linear(in_dim=10, out_dim=5, bias=False)
 67        >>>
 68        >>> # Linear layer without activation
 69        >>> layer = Linear(in_dim=10, out_dim=5, act_fn=None)
 70    """
 71
 72    @property
 73    def in_dim(self) -> int:
 74        """Input dimension of the layer."""
 75        return self._in_dim
 76
 77    @property
 78    def out_dim(self) -> int:
 79        """Output dimension of the layer."""
 80        return self._out_dim
 81
 82    @property
 83    def hyperparameters(self) -> dict:
 84        """Dictionary containing the layer hyperparameters."""
 85        return {
 86            "layer": "Linear",
 87            "in_dim": self._in_dim,
 88            "out_dim": self._out_dim,
 89            "act_fn": str(self._act_fn) if self._act_fn is not None else None,
 90            "optimizer": {
 91                "cache": self._optimizer.cache if self._optimizer is not None else {},
 92                "hyperparameters": (self._optimizer.hyperparameters if self._optimizer is not None else {}),
 93            },
 94        }
 95
 96    def __init__(
 97        self,
 98        in_dim: int,
 99        out_dim: int,
100        bias: bool = True,
101        act_fn: str = None,
102    ):
103        """
104        Initialize a Linear layer.
105
106        Args:
107            in_dim (int): Input dimension.
108            out_dim (int): Output dimension.
109            bias (bool, optional): Include bias term. Defaults to True.
110            act_fn (str, optional): Activation function. Defaults to None.
111        """
112        super(Linear, self).__init__()
113        self._in_dim = in_dim
114        self._out_dim = out_dim
115        self._bias = bias
116        self._act_fn = ActivationInitializer(act_fn)()
117        self._parameters = {"W": None, "b": None} if bias else {"W": None}
118        self._gradients = {"W": None, "b": None} if bias else {"W": None}
119        self._derived_variables = {"z": None}
120        self._init = False
121
122    def __str__(self) -> str:
123        """Return a string representation of the layer."""
124        return (
125            f"Layer        | Linear(in_dim={self._in_dim}, out_dim={self._out_dim}) \n"
126            + f"n_params     | {self._out_dim * (self._in_dim + 1) if self._bias else self._out_dim * self._in_dim} \n"
127            + f"act_fn       | {str(self._act_fn)} \n"
128            + f"optimizer    | {str(self._optimizer)} \n"
129        )
130
131    def set_activation(self, act_fn: str) -> None:
132        """
133        Set or change the activation function.
134
135        Args:
136            act_fn (str): Name of the activation function.
137                Options: "sigmoid", "relu", "tanh", "softplus", None.
138
139        Examples:
140            >>> layer.set_activation("relu")
141            >>> layer.set_activation(None)  # No activation
142        """
143        self._act_fn = ActivationInitializer(act_fn)()
144
145    def init_params(self) -> None:
146        """
147        Initialize the layer's trainable parameters.
148
149        Uses Kaiming uniform initialization for weights and He uniform
150        for biases. The initialization scales with 1/sqrt(in_dim).
151
152        Examples:
153            >>> layer.init_params()
154            >>> print(layer._parameters["W"].shape)
155            (5, 10)
156        """
157        # Kaiming uniform initialization for weights
158        b = np.sqrt(1 / self._in_dim)
159        W = np.random.uniform(-b, b, size=(self._out_dim, self._in_dim))
160        self._parameters["W"] = W
161        self._gradients["W"] = np.zeros_like(W)
162        if self._bias:
163            bias = np.random.uniform(-b, b, size=(1, self._out_dim))
164            self._parameters["b"] = bias
165            self._gradients["b"] = np.zeros_like(bias)
166        self._init = True

168    def forward(self, X: np.ndarray, retain_derived: bool = True) -> np.ndarray:
169        """
170        Perform forward propagation.
171
172        Computes: z = X @ W^T + b, y = activation(z)
173
174        Args:
175            X (np.ndarray): Input data of shape (batch_size, in_dim)
176                or (in_dim,) for a single sample.
177            retain_derived (bool, optional): Whether to retain intermediate
178                values for backward propagation. Defaults to True.
179
180        Returns:
181            np.ndarray: Output of shape (batch_size, out_dim).
182
183        Raises:
184            ValueError: If input dimensions don't match.
185

187            >>> X = np.random.randn(32, 10)
188            >>> output = layer.forward(X)  # Shape: (32, 5)
189            >>>

192            >>> output = layer.forward(X)  # Shape: (5,)

194        if X is None:
195            raise ValueError("Input should not be None.")
196        if X.ndim == 1:
197            X = X.reshape(1, -1)
198        if X.shape[1] != self._in_dim:
199            raise ValueError(f"Input dimension {X.shape[1]} does not match expected {self._in_dim}.")
200        if not self._init:
201            self.init_params()
202        W = self._parameters["W"]
203        b = self._parameters["b"] if self._bias else 0
204        z = np.dot(X, W.T) + b
205        y = self._act_fn(z) if self._act_fn else z

207        if retain_derived:
208            self._X = X
209            self._derived_variables["z"] = z

213    def _bwd(self, dLdy: np.ndarray, x: np.ndarray) -> tuple:
214        """
215        Compute gradients for a single sample.
216
217        Args:
218            dLdy (np.ndarray): Gradients from the next layer.
219            x (np.ndarray): Input for this sample.
220
221        Returns:
222            tuple: (dLdx, dLdw, dLdb) where:
223                - dLdx: Gradients for the input
224                - dLdw: Gradients for the weights
225                - dLdb: Gradients for the bias
226        """
227        W = self._parameters["W"]
228        b = self._parameters["b"] if self._bias else 0
229        z = np.dot(x, W.T) + b
230
231        dydz = self._act_fn.grad(z) if self._act_fn else 1
232        dLdz = dLdy * dydz
233        dLdx = dLdz @ W
234        dLdw = dLdz.T @ x
235        dLdb = dLdz.sum(axis=0, keepdims=True)
236
237        return dLdx[0], dLdw, dLdb

240        """
241        Perform backward propagation.
242
243        Computes gradients of the loss with respect to the input,
244        weights, and bias using the chain rule.
245
246        Args:
247            dLdy (np.ndarray): Gradients of the loss with respect to
248                the layer output. Shape: (batch_size, out_dim) or (out_dim,).
249            retain_grad (bool, optional): Whether to retain gradients
250                for parameter updates. Defaults to True.
251
252        Returns:
253            np.ndarray: Gradients with respect to the input.
254                Shape: (batch_size, in_dim) or (in_dim,) for single sample.
255
256        Raises:
257            ValueError: If layer is frozen or dimensions mismatch.
258
259        Examples:
260            >>> dLdy = np.random.randn(32, 5)
261            >>> dX = layer.backward(dLdy)  # Shape: (32, 10)
262            >>>
263            >>> # Single sample
264            >>> dLdy = np.random.randn(5)
265            >>> dX = layer.backward(dLdy)  # Shape: (10,)
266        """
267        if not self._trainable:
268            raise ValueError("Layer is frozen.")
269        if isinstance(dLdy, numbers.Number):
270            dLdy = np.array([dLdy])
271        if not isinstance(dLdy, np.ndarray):
272            raise TypeError(f"Expected np.ndarray, got {type(dLdy).__name__}")
273        if dLdy.ndim == 1:
274            dLdy = dLdy.reshape(1, -1)
275        if dLdy.shape[1] != self._out_dim:
276            raise ValueError(f"Gradient dimension {dLdy.shape[1]} does not match output dimension {self._out_dim}.")
277        if self._X.shape[0] != dLdy.shape[0]:
278            raise ValueError(f"Batch size mismatch: input batch {self._X.shape[0]} vs gradient batch {dLdy.shape[0]}.")
279
280        dX = []
281        X = self._X
282        for dy, x in zip(dLdy, X):
283            if x.ndim == 1:
284                x = x.reshape(1, -1)
285            if dy.ndim == 1:
286                dy = dy.reshape(1, -1)
287            dx, dw, db = self._bwd(dy, x)
288            dX.append(dx)
289            if retain_grad:
290                self._gradients["W"] += dw
291                if self._bias:
292                    self._gradients["b"] += db
293
294        return dX[0] if len(X) == 1 else np.array(dX)
