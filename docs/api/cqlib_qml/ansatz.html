cqlib_qml.ansatz

Parameterized quantum circuits (ansatz) for variational quantum algorithms.

This module provides a collection of parameterized quantum circuit architectures (ansatz) for use in variational quantum algorithms, quantum neural networks, and other quantum machine learning models.

The ansatz is a key component that defines the structure of the parameterized quantum circuit. Different ansatz architectures offer different trade-offs between expressivity, trainability, and hardware efficiency.

Available Ansatz: - Ansatz: Base class for all ansatz implementations. - BasicQNN: Basic QNN ansatz with alternating 2-qubit gates. - CRADL: Color-Readout-Alternating-Double-Layer architecture. - CRAML: Color-Readout-Alternating-Mixed-Layer architecture. - HEAnsatz: Hardware-Efficient ansatz with alternating single and 2-qubit gates.

Examples:

from cqlib_qml.ansatz import HEAnsatz, BasicQNN, CRADL

Hardware-Efficient ansatz

he = HEAnsatz(n_qubits=4, d=3, layers=["RY", "CX"])

Basic QNN ansatz

basic = BasicQNN(n_qubits=4, layers=["XX", "ZZ"])

CRADL ansatz for image data

cradl = CRADL(n_qubits=5, layers=2)

 219    def add_encoder(self, other: Union[Circuit, List[Circuit], np.ndarray]) -> None:
 220        """
 221        Add encoding circuits to the ansatz.
 222
 223        This method attaches encoding circuits that will be prepended to the
 224        ansatz during forward propagation.
 225
 226        Args:
 227            other: Encoding circuit(s) to add. Can be a single Circuit,
 228                a list of Circuits, or a numpy array of Circuits.
 229
 230        Raises:
 231            TypeError: If any element is not a Circuit.
 232            ValueError: If any encoding circuit contains unassigned parameters
 233                or if widths are inconsistent.
 234
 235        Examples:
 236            >>> from cqlib_qml.encoder import FRQI
 237            >>> encoder = FRQI(n_pixels=4)
 238            >>> data = np.array([[0, 1], [1, 0]])
 239            >>> data_circuits = encoder(data)
 240            >>> ansatz.add_encoder(data_circuits)
 241        """
 242        enc_cirs = [other] if isinstance(other, Circuit) else other
 243        if isinstance(enc_cirs, np.ndarray):
 244            enc_cirs = enc_cirs.tolist()
 245        if isinstance(enc_cirs, list):
 246            width = enc_cirs[0].width
 247            for enc_cir in enc_cirs:
 248                if not isinstance(enc_cir, Circuit):
 249                    raise TypeError(f"Expected Circuit, got {type(enc_cir).__name__}")
 250                if len(enc_cir.parameters) > 0:
 251                    raise ValueError("The encoder circuit cannot contain unassigned parameters.")
 252                if enc_cir.width != width:
 253                    raise ValueError("Width of all encoding circuits must be consistent.")
 254        else:
 255            raise TypeError(f"Expected List[Circuit], got {type(enc_cirs).__name__}")
 256        self._encoder = enc_cirs

 278    def _get_fwd_circuits(self, X: Optional[np.ndarray] = None) -> tuple:
 279        """
 280        Generate forward circuits with parameter bindings.
 281
 282        Args:
 283            X: Input data for parameter binding. If None, uses random
 284                initialization or existing bindings.
 285
 286        Returns:
 287            tuple: (circuits, bindings_list) where circuits is a list of
 288                Circuit objects and bindings_list is a list of bindings.
 289        """
 290        circuits = []
 291        # Parameters need to be updated
 292        if X is None:
 293            # random initialization
 294            if not self._assigned_value():
 295                keys = self.symbols
 296                values = np.random.randn(len(self.symbols))
 297                self._bindings = dict(zip(keys, values))
 298                self._assigned_cir = self._circuit.assign_parameters(self._bindings)
 299            # encoder + ansatz
 300            if self._encoder is not None:
 301                for enc_cir in self._encoder:
 302                    enc_cir_gate = enc_cir.to_gate("enc_cir")
 303                    enc_cir_copy = Circuit(enc_cir.num_qubits)
 304                    enc_cir_copy.append(enc_cir_gate, list(range(enc_cir.num_qubits)))
 305                    enc_cir_copy.compose(self._assigned_cir)
 306                    circuits.append(enc_cir_copy)
 307            else:
 308                circuits = [self._assigned_cir]
 309            bindings_list = [self._bindings]
 310        # Parameters depend on the previous layer
 311        else:
 312            bindings_list = []
 313            self._updatable = False
 314            if X.ndim == 1:
 315                X = X.reshape(1, -1)
 316            if X.shape[1] != self.in_dim:
 317                raise ValueError(f"Input dimension {X.shape[1]} does not match expected {self.in_dim}.")
 318            # X + encoder + ansatz
 319            if self._encoder is not None:
 320                if len(self._encoder) != X.shape[0]:
 321                    raise ValueError(f"Batch size {X.shape[0]} does not match encoding circuits {len(self._encoder)}.")
 322                for x, enc_cir in zip(X, self._encoder):
 323                    bindings = dict(zip(self.symbols, x))
 324                    assigned_cir = self._circuit.assign_parameters(bindings)
 325                    enc_cir_gate = enc_cir.to_gate("enc_cir")
 326                    enc_cir_copy = Circuit(enc_cir.num_qubits)
 327                    enc_cir_copy.append(enc_cir_gate, list(range(enc_cir.num_qubits)))
 328                    enc_cir_copy.compose(assigned_cir)
 329                    circuits.append(enc_cir_copy)
 330                    bindings_list.append(bindings)
 331            # X + ansatz
 332            else:
 333                for x in X:
 334                    bindings = dict(zip(self.symbols, x))
 335                    assigned_cir = self._circuit.assign_parameters(bindings)
 336                    circuits.append(assigned_cir)
 337                    bindings_list.append(bindings)
 338        return circuits, bindings_list

 340    def _fwd(
 341        self, circuits: List[Circuit], quantum_state: Optional[np.ndarray], bindings_list: List[Dict]
 342    ) -> np.ndarray:
 343        """
 344        Execute forward pass on the circuits.
 345
 346        Args:
 347            circuits: List of circuits to execute.
 348            quantum_state: Optional initial quantum state.
 349            bindings_list: List of parameter bindings.
 350
 351        Returns:
 352            np.ndarray: Expectation values.
 353        """
 354        n_circuits = len(circuits)
 355        n_states = quantum_state.shape[0] if quantum_state is not None else 0
 356        if not (n_circuits == n_states or n_states in [0, 1] or n_circuits == 1):
 357            raise ValueError(f"Circuit batch size {n_circuits} does not match state batch size {n_states}.")
 358        n = max(n_circuits, n_states)
 359        expectations = []
 360        for i in range(n):
 361            if n_states == 0:
 362                state_vector = None
 363            else:
 364                state_vector = quantum_state[i] if n_states == n else quantum_state[0]
 365            cir = circuits[i] if n_circuits == n else circuits[0]
 366            if self._readouts is not None:
 367                exps, sv = self._get_pauliZ_expectations(cir, self._readouts, state_vector)
 368            else:
 369                exps, sv = self._get_expectations(cir, self._hams, state_vector)
 370            expectations.append(exps)
 371            if self._trainable:
 372                self._bwd(sv) if self._updatable else self._bwd(sv, bindings_list[i])
 373
 374        return expectations[0] if len(expectations) == 0 else np.array(expectations)

 376    def forward(
 377        self, X: Optional[np.ndarray] = None, quantum_state: Optional[np.ndarray] = None
 378    ) -> Union[float, np.ndarray]:
 379        """
 380        Perform forward propagation for one step.
 381
 382        This method computes the expectation values of the quantum circuit
 383        with the given parameters and optional input data.
 384
 385        Args:
 386            X (np.ndarray, optional): Input data for parameter binding.
 387                If None, uses random initialization. Defaults to None.
 388            quantum_state (np.ndarray, optional): Initial quantum state vector.
 389                Defaults to None (|0⟩ state).
 390
 391        Returns:
 392            Union[float, np.ndarray]: Expectation values of the measurements.
 393
 394        Raises:
 395            ValueError: If circuit is not initialized or input dimensions mismatch.
 396
 397        Examples:
 398            >>> ansatz = BasicQNN(n_qubits=3, layers=["XX"])
 399            >>> ansatz.set_measurement(readouts=[0])
 400            >>> result = ansatz.forward()  # Random initialization
 401            >>> print(result.shape)
 402            (1, 1)
 403            >>>
 404            >>> # With specific parameters
 405            >>> ansatz.assign_parameters({"theta": 0.5})
 406            >>> result = ansatz.forward()
 407        """
 408        if len(self) == 0:
 409            raise ValueError("The circuit must be initialized.")
 410        if self._differentiator is None:
 411            self._differentiator = AdjointDifferentiator()
 412        if self._optimizer is None:
 413            self.set_optimizer()
 414        if quantum_state is not None:
 415            quantum_state = self._validate_vectors(quantum_state, self.num_qubits)
 416        circuits, bindings_list = self._get_fwd_circuits(X)

 420    def _bwd(self, sv: np.ndarray, bindings: Optional[Dict] = None) -> None:
 421        """
 422        Compute gradients using the differentiator.
 423
 424        Args:
 425            sv: Statevector from forward pass.
 426            bindings: Parameter bindings. If None, uses stored bindings.
 427        """
 428        bindings = self._bindings if bindings is None else bindings
 429        if isinstance(self._differentiator, AdjointDifferentiator):
 430            grads = self._differentiator.run(self._circuit, bindings, sv, self._readouts, self._hams)
 431        else:
 432            grads = self._differentiator.run(self._circuit, bindings, self._readouts, self._hams)
 433        if len(self._gradients) == 0:
 434            self._gradients = grads.copy()
 435        else:
 436            for key in grads.keys():
 437                self._gradients[key] = np.vstack((self._gradients[key], grads[key]))

 439    def backward(self, dLdexp: Optional[Union[float, np.ndarray]] = None) -> Union[dict, np.ndarray]:
 440        """
 441        Perform backward propagation for one step.
 442
 443        This method computes the gradients of the loss with respect to the
 444        trainable parameters or inputs.
 445
 446        Args:
 447            dLdexp (Union[float, np.ndarray], optional): Gradients of the loss
 448                with respect to the expectation values. If None, returns the
 449                computed gradients. Defaults to None.
 450
 451        Returns:
 452            Union[dict, np.ndarray]:
 453                - If dLdexp is None: Returns the gradients dictionary.
 454                - If dLdexp is provided: Returns the gradients of inputs.
 455
 456        Raises:
 457            ValueError: If the ansatz is frozen or dimensions mismatch.
 458
 459        Examples:
 460            >>> # Forward pass
 461            >>> expectations = ansatz.forward(data_circuits)
 462            >>>
 463            >>> # Compute loss
 464            >>> loss = loss_fun(expectations, labels)
 465            >>>
 466            >>> # Backward pass
 467            >>> ansatz.backward(loss_fun.grads())
 468        """
 469        if not self._trainable:
 470            raise ValueError("Ansatz is frozen.")
 471        if dLdexp is None:
 472            return self._gradients
 473        else:
 474            if isinstance(dLdexp, numbers.Number):
 475                dLdexp = np.array([dLdexp])
 476            if not isinstance(dLdexp, np.ndarray):
 477                raise TypeError(f"Expected np.ndarray, got {type(dLdexp).__name__}")
 478            if dLdexp.ndim == 1:
 479                dLdexp = dLdexp.reshape(1, -1)
 480            if dLdexp.shape[1] != self._out_dim:
 481                raise ValueError(
 482                    f"Gradient dimension {dLdexp.shape[1]} does not match output dimension {self._out_dim}."
 483                )
 484            expected_batch_size = len(self._encoder) if self._encoder is not None else 1
 485            if dLdexp.shape[0] != expected_batch_size:
 486                raise ValueError(
 487                    f"Gradient batch size {dLdexp.shape[0]} does not match encoding batch {expected_batch_size}."
 488                )
 489
 490            for key, val in self._gradients.items():
 491                if val.ndim == 1:
 492                    val = val.reshape(1, -1)
 493                if val.shape != dLdexp.shape:
 494                    raise ValueError(f"Gradient shape {val.shape} does not match dLdexp shape {dLdexp.shape}.")
 495                if self._updatable:
 496                    self._gradients[key] = sum(sum(val * dLdexp))
 497                else:
 498                    self._gradients[key] = np.sum(val * dLdexp, axis=1)
 499            if self._updatable:
 500                return self._gradients
 501            else:
 502                gradients = None
 503                for val in self._gradients.values():
 504                    if gradients is None:
 505                        gradients = val.copy()
 506                    else:
 507                        gradients = np.column_stack((gradients, val))
 508                return gradients

 562    def set_measurement(self, **kwargs) -> None:
 563        """
 564        Set the measurement for the model.
 565
 566        Either `readouts` or `hams` must be provided. If both are provided,
 567        `readouts` takes precedence.
 568
 569        Args:
 570            readouts (Union[int, list]): Qubit indices for Pauli-Z measurement.
 571            hams (Union[Hamiltonian, list]): Hamiltonians for expectation calculation.
 572
 573        Raises:
 574            ValueError: If both or neither are provided.
 575
 576        Examples:
 577            >>> # Measure qubits 0 and 1
 578            >>> ansatz.set_measurement(readouts=[0, 1])
 579            >>>
 580            >>> # Measure custom Hamiltonian
 581            >>> from cqlib.qis import Hamiltonian, PauliString
 582            >>> ham = Hamiltonian(2)
 583            >>> ham.add_term(PauliString.from_str("ZZ"), 1.0)
 584            >>> ansatz.set_measurement(hams=[ham])
 585        """
 586        kwargs_keys = list(kwargs.keys())
 587        if len(kwargs_keys) != 1:
 588            raise ValueError("Only one of 'readouts' and 'hams' is accepted as the key.")
 589        kwargs_key = kwargs_keys[0]
 590        kwargs_val = list(kwargs.values())[0]
 591        if kwargs_key == "readouts":

 614        Update the trainable parameters according to gradients.

 616        Args:
 617            cur_loss (float, optional): Current loss value for learning rate

 619

 621            ValueError: If the ansatz is frozen.
 622
 623        Examples:
 624            >>> ansatz.forward()

 634                new_params[k] = self._optimizer(self._bindings[k], v, k, cur_loss)

 689    def _circuit_summary(self) -> dict:
 690        """Generate a summary of the circuit structure."""
 691        if len(self) == 0:
 692            return {}
 693        gates = []
 694        for op in self.operations:
 695            instruction = op.instruction
 696            gate_info = {}
 697            gate_info["qubits"] = [qubit.index for qubit in op.qubits]
 698            gate_info["num_qubits"] = op.num_qubits
 699            params = []
 700            for param in op.params:
 701                if isinstance(param, tuple):
 702                    param_name = str(self.parameters[param[1]])
 703                    params.append(param_name)
 704                else:
 705                    params.append(param)
 706            gate_info["params"] = params
 707
 708            if instruction.is_standard:
 709                gate_info["type"] = "standard"
 710                gate_info["name"] = str(instruction.standard_gate)
 711            elif instruction.is_mcgate:
 712                mc_gate = instruction.mc_gate
 713                gate_info["type"] = "mcgate"
 714                gate_info["base_gate"] = str(mc_gate.base_gate)
 715                gate_info["num_ctrl_qubits"] = mc_gate.num_ctrl_qubits
 716            elif instruction.is_unitary:
 717                unitary_gate = instruction.unitary_gate
 718                gate_info["type"] = "unitary"
 719                gate_info["label"] = unitary_gate.label
 720                gate_info["matrix"] = unitary_gate.matrix
 721            else:
 722                pass
 723            gates.append(gate_info)
 724        return {"num_qubits": self.num_qubits, "parameters": self._bindings, "gates": gates}

 726    def _load_circuit(self, cir_summary: dict) -> None:
 727        """Load a circuit from a summary dictionary."""
 728        if len(self) != 0:
 729            raise ValueError("Needs to be an empty circuit.")
 730        num_qubits = cir_summary["num_qubits"]
 731        if self.num_qubits == 0:
 732            self.add_qubits(list(range(num_qubits)))
 733        else:
 734            if self.num_qubits != num_qubits:
 735                raise ValueError("The loaded circuit does not match the width of current circuit.")
 736        for gate_info in cir_summary["gates"]:
 737            gate_qubits = gate_info["qubits"]
 738            gate_num_qubits = gate_info["num_qubits"]
 739            gate_params = []
 740            for param in gate_info["params"]:
 741                if isinstance(param, str):
 742                    param = Parameter(param)
 743                gate_params.append(param)
 744            gate_type = gate_info["type"]

 757    def _validate_hamiltonians(self, hams):
 758        """Validate and format Hamiltonians."""
 759        if isinstance(hams, Hamiltonian):
 760            hams = [hams]
 761        if isinstance(hams, list):
 762            self._out_dim = len(hams)
 763            for ham in hams:
 764                if not isinstance(ham, Hamiltonian):
 765                    raise TypeError(f"Expected Hamiltonian, got {type(ham).__name__}")
 766        else:
 767            raise TypeError(f"Expected Hamiltonian or List[Hamiltonian], got {type(hams).__name__}")
 768        return hams

 770    def _validate_readouts(self, readouts):
 771        """Validate and format readouts."""
 772        if isinstance(readouts, int):
 773            readouts = [readouts]
 774        if isinstance(readouts, list):
 775            if len(set(readouts)) != len(readouts):
 776                raise ValueError("readout must be a list of non-repeating integers.")
 777            self._out_dim = len(readouts)
 778            for readout in readouts:
 779                if not isinstance(readout, int):
 780                    raise TypeError(f"Expected int, got {type(readout).__name__}")
 781                if readout < 0 or readout >= self.num_qubits:
 782                    raise ValueError(f"Readout {readout} must be in range [0, {self.num_qubits - 1}].")
 783        else:
 784            raise TypeError(f"Expected int or List[int], got {type(readouts).__name__}")
 785        return readouts

 787    def _readouts2hams(self, readouts: list) -> List[Hamiltonian]:
 788        """Convert readouts to Pauli-Z Hamiltonians."""
 789        hams = []
 790        for readout in readouts:
 791            ham = Hamiltonian(self.num_qubits)
 792            pauli = "".join(["Z" if i == self.num_qubits - 1 - readout else "I" for i in range(self.num_qubits)])
 793            ham.add_term(PauliString.from_str(pauli), 1)
 794            hams.append(ham)
 795        return hams

Base class for all parameterized quantum circuit ansätze.

This class provides the foundation for building parameterized quantum circuits with support for automatic differentiation, parameter updates, and integration with classical neural network components.

The Ansatz class manages: - Quantum circuit construction and parameter assignment - Forward propagation (expectation value computation) - Backward propagation (gradient computation) - Parameter optimization - State saving and loading

Attributes: in_dim (int): Number of input dimensions (trainable parameters). out_dim (int): Number of output dimensions (measurements). gradients (dict): Gradients of trainable parameters. readouts (list): Readout qubits for measurement. hams (list): Hamiltonians for expectation calculation. trainable (bool): Whether the ansatz is trainable. updatable (bool): Whether parameters can be updated. num_qubits (int): Number of qubits in the circuit. width (int): Alias for num_qubits. qubits (list): List of all qubits in the circuit. parameters (list): List of symbolic parameters. symbols (list): Names of symbolic parameters.

Note: User-defined ansatz classes should inherit from this base class.

Examples:

Create a simple ansatz

ansatz = Ansatz(2) theta = Parameter("theta") ansatz.ry(0, theta) ansatz.cx(0, 1) ansatz.set_measurement(readouts=[0])

Forward pass

result = ansatz.forward()

Assign specific parameters

ansatz.assign_parameters({"theta": 0.5})

def backward( self, dLdexp: Union[float, numpy.ndarray, NoneType] = None) -> Union[dict, numpy.ndarray]: 
439    def backward(self, dLdexp: Optional[Union[float, np.ndarray]] = None) -> Union[dict, np.ndarray]:
440        """
441        Perform backward propagation for one step.
442
443        This method computes the gradients of the loss with respect to the
444        trainable parameters or inputs.
445
446        Args:
447            dLdexp (Union[float, np.ndarray], optional): Gradients of the loss
448                with respect to the expectation values. If None, returns the
449                computed gradients. Defaults to None.
450
451        Returns:
452            Union[dict, np.ndarray]:
453                - If dLdexp is None: Returns the gradients dictionary.
454                - If dLdexp is provided: Returns the gradients of inputs.
455
456        Raises:
457            ValueError: If the ansatz is frozen or dimensions mismatch.
458
459        Examples:
460            >>> # Forward pass
461            >>> expectations = ansatz.forward(data_circuits)
462            >>>
463            >>> # Compute loss
464            >>> loss = loss_fun(expectations, labels)
465            >>>
466            >>> # Backward pass
467            >>> ansatz.backward(loss_fun.grads())
468        """
469        if not self._trainable:
470            raise ValueError("Ansatz is frozen.")
471        if dLdexp is None:
472            return self._gradients
473        else:
474            if isinstance(dLdexp, numbers.Number):
475                dLdexp = np.array([dLdexp])
476            if not isinstance(dLdexp, np.ndarray):
477                raise TypeError(f"Expected np.ndarray, got {type(dLdexp).__name__}")
478            if dLdexp.ndim == 1:
479                dLdexp = dLdexp.reshape(1, -1)
480            if dLdexp.shape[1] != self._out_dim:
481                raise ValueError(
482                    f"Gradient dimension {dLdexp.shape[1]} does not match output dimension {self._out_dim}."
483                )
484            expected_batch_size = len(self._encoder) if self._encoder is not None else 1
485            if dLdexp.shape[0] != expected_batch_size:
486                raise ValueError(
487                    f"Gradient batch size {dLdexp.shape[0]} does not match encoding batch {expected_batch_size}."
488                )
489
490            for key, val in self._gradients.items():
491                if val.ndim == 1:
492                    val = val.reshape(1, -1)
493                if val.shape != dLdexp.shape:
494                    raise ValueError(f"Gradient shape {val.shape} does not match dLdexp shape {dLdexp.shape}.")
495                if self._updatable:
496                    self._gradients[key] = sum(sum(val * dLdexp))
497                else:
498                    self._gradients[key] = np.sum(val * dLdexp, axis=1)
499            if self._updatable:
500                return self._gradients
501            else:
502                gradients = None
503                for val in self._gradients.values():
504                    if gradients is None:
505                        gradients = val.copy()
506                    else:
507                        gradients = np.column_stack((gradients, val))
508                return gradients
